การสร้าง Server ด้วย Node.js ล้วน ๆ

เหลือคำสั่งสุดท้ายคือ web เราจะให้คำสั่งนี้เปิดเว็บเซิร์ฟเวอร์และแสดงรายการ Todo ทั้งหมดใน Browser

งานนี้ใช้เพียง Module http ที่ติดมากับ Node.js ไม่ต้องติดตั้งแพ็กเกจเพิ่ม

สร้างไฟล์ server.js

ให้สร้างไฟล์ server.js ในโฟลเดอร์ src:

import http from 'node:http'

const buildHtml = (todos) => {
  const items = todos.map(todo => {
    const mark = todo.done ? '✅' : '⬜'
    return `<li>${mark} ${todo.text}</li>`
  }).join('')

  return `
    <h1>รายการสิ่งที่ต้องทำ</h1>
    <ul>${items}</ul>
  `
}

export const start = (todos, port) => {
  const server = http.createServer((req, res) => {
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' })
    res.end(buildHtml(todos))
  })

  server.listen(port, () => {
    console.log(`เปิดดูได้ที่ http://localhost:${port}`)
  })
}

คำอธิบายโค้ด

  • buildHtml รับ Array ของ Todo แล้วสร้างข้อความ HTML ขึ้นมา เราใช้ map แปลงแต่ละรายการให้เป็น <li> แล้วใช้ join('') เชื่อมทุกตัวติดกันเป็นข้อความเดียว ถ้าไม่ใส่ join เวลา Array ถูกแปลงเป็นข้อความมันจะมีเครื่องหมายคอมมาคั่นให้เห็นด้วย
  • http.createServer รับฟังก์ชันเข้าไปหนึ่งตัว ซึ่งจะถูกเรียกทุกครั้งที่มี Request เข้ามา ฟังก์ชันนี้ได้ req คือข้อมูลของ Request ที่เข้ามา และ res คือตัวที่เราใช้ตอบกลับ
  • res.writeHead(200, ...) คือการบอกสถานะว่าสำเร็จ (200) พร้อมบอกชนิดข้อมูลว่าเป็น HTML ส่วน charset=utf-8 สำคัญมาก ถ้าไม่ใส่ ภาษาไทยจะแสดงผลเป็นตัวอักษรแปลก ๆ ใน Browser
  • res.end(...) คือการส่งเนื้อหากลับไปแล้วปิด Response
  • server.listen(port, ...) คือการเริ่มรอรับ Request ที่ Port ที่ระบุ ฟังก์ชันข้างในจะทำงานเมื่อเซิร์ฟเวอร์พร้อมแล้ว เราจึงใช้ตรงนี้แสดง URL ให้ผู้ใช้กดเข้าไปดู

สังเกตว่าเซิร์ฟเวอร์ของเราตอบ HTML เดิมกลับไปเสมอไม่ว่าจะขอ Path ไหนก็ตาม เพราะเรายังไม่ได้ดู req.url เลย ซึ่งพอสำหรับแอปเล็ก ๆ แบบนี้ ส่วนการแยกเส้นทางตาม URL เป็นเรื่องที่เราจะได้เรียนกันต่อไปตอนใช้ Framework อย่าง Express

นำไปใช้กับคำสั่ง web

กลับไปที่ index.js เพิ่ม Import ตัวใหม่ไว้ข้างบน:

import { start } from './src/server.js'

แล้วเพิ่ม case ใหม่เข้าไปใน switch:

  case 'web': {
    start(todos, Number(arg) || 3000)
    break
  }

ส่วน || 3000 คือการกำหนดค่าเริ่มต้น ถ้าผู้ใช้ไม่ได้ระบุ Port มา arg จะเป็น undefined และ Number(undefined) จะได้ NaN ซึ่งเป็นค่าที่ JavaScript ถือว่าเป็นเท็จ ทำให้ || เลือกใช้ 3000 แทน

ทดลองใช้งาน

todo web

จะได้ผลลัพธ์:

เปิดดูได้ที่ http://localhost:3000

กด Link หรือคัดลอกไปเปิดใน Browser ก็จะเห็นรายการ Todo ทั้งหมด หากอยากใช้ Port อื่นให้ระบุต่อท้าย เช่น todo web 4000

ถ้าเจอ Error ว่า EADDRINUSE

ถ้ารันแล้วเจอข้อความประมาณนี้:

Error: listen EADDRINUSE: address already in use :::3000

ข้อความนี้แปลว่ามีโปรแกรมอื่นใช้ Port นั้นอยู่แล้ว เพราะหนึ่ง Port เปิดรับฟังได้ทีละโปรแกรม วิธีแก้ง่ายที่สุดคือเปลี่ยนเลข เช่น todo web 4000

สาเหตุที่พบบ่อยคือเราเปิด Dev Server ของโปรเจกต์ React ค้างไว้อยู่ ซึ่งก็ใช้ Port 3000 เหมือนกัน

อีกกรณีหนึ่งที่คนใช้ Mac มักเจอคือ Port 5000 ถูกฟีเจอร์ AirPlay Receiver ของ macOS จองไว้ตลอดเวลา ดังนั้นถ้าคุณลองเปลี่ยนไปใช้ 5000 แล้วยังไม่ได้ อย่าเพิ่งคิดว่าโค้ดตัวเองผิด ให้เปลี่ยนไปใช้เลขอื่นแทน

สังเกตว่าคำสั่งนี้จะไม่จบเอง เพราะเซิร์ฟเวอร์ต้องคอยรอรับ Request อยู่ตลอด ถ้าอยากหยุดให้กด Ctrl + C ใน Terminal

ข้อสังเกตอีกอย่างคือหน้าเว็บจะไม่อัปเดตเองถ้าเราไปเพิ่มรายการใหม่จาก Terminal อีกหน้าต่างหนึ่ง เพราะเราอ่านข้อมูลจากไฟล์แค่ครั้งเดียวตอนเริ่มเซิร์ฟเวอร์แล้วส่งเข้าไปให้ start ถ้าอยากให้ข้อมูลสดเสมอ ลองคิดดูว่าต้องย้ายการเรียก readTodos ไปไว้ตรงไหน ถือเป็นแบบฝึกหัดเล็ก ๆ ท้ายบทนะ
