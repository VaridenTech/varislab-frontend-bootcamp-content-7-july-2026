การอัปเดตคำสั่งให้ทำงานได้จริง

ตอนนี้ชิ้นส่วนพร้อมแล้ว ทั้งฟังก์ชันอ่านเขียนไฟล์และฟังก์ชันจัดการ Array เหลือเพียงต่อทุกอย่างเข้ากับ switch ใน index.js ที่วางโครงไว้ตั้งแต่บทที่ 15

โค้ดฉบับสมบูรณ์ของ index.js

#!/usr/bin/env node
import { readTodos, saveTodos, addTodo, markDone, removeTodo } from './src/todos.js'

const [command, arg] = process.argv.slice(2)
const todos = await readTodos()

switch (command) {
  case 'add': {
    await saveTodos(addTodo(todos, arg))
    console.log('เพิ่มแล้ว:', arg)
    break
  }
  case 'list': {
    todos.forEach(todo => {
      console.log(todo.id, todo.done ? '✅' : '⬜', todo.text)
    })
    break
  }
  case 'done': {
    await saveTodos(markDone(todos, Number(arg)))
    console.log('ทำเสร็จแล้ว:', arg)
    break
  }
  case 'remove': {
    await saveTodos(removeTodo(todos, Number(arg)))
    console.log('ลบแล้ว:', arg)
    break
  }
  default: {
    console.log('คำสั่งที่ใช้ได้: add, list, done, remove, web')
  }
}

คำอธิบายโค้ด

รูปแบบของทุกคำสั่งที่แก้ไขข้อมูลจะเหมือนกันหมด คือ อ่านเข้ามา → แปลง → เขียนกลับ ตัวอย่างเช่นบรรทัดนี้:

await saveTodos(addTodo(todos, arg))

อ่านจากในออกนอก: addTodo(todos, arg) คืน Array ใหม่ที่มีรายการเพิ่มเข้ามา แล้วเราส่ง Array นั้นให้ saveTodos เขียนลงไฟล์ทันที

ระวัง Number(arg) ให้ดี เพราะค่าจาก process.argv เป็น String เสมอ เมื่อพิมพ์ todo done 3 เราจะได้ '3' ไม่ใช่ตัวเลข 3 และ === ตรวจชนิดข้อมูลด้วย จึงทำให้ '3' === 3 เป็น false ถ้าลืมแปลง คำสั่ง done กับ remove จะเงียบและไม่แก้ข้อมูล แม้ไม่มี Error แสดงขึ้นมา

ส่วนคำสั่ง list ไม่ต้องบันทึกอะไร แค่วนแสดงผลอย่างเดียว โดยใช้ Ternary เลือกสัญลักษณ์ ✅ หรือ ⬜ ตามสถานะ done

สังเกตว่าเราใช้ await ที่ระดับบนสุดของไฟล์ได้เลยโดยไม่ต้องห่อด้วย async function ความสามารถนี้เรียกว่า Top-level await ซึ่งใช้ได้เพราะเราตั้ง "type": "module" ไว้ใน package.json ตั้งแต่บทที่ 9

ทดลองใช้งาน

ลองรันตามลำดับนี้ดู:

todo add "ซื้อนม"
todo add "อ่านหนังสือ"
todo add "ออกกำลังกาย"
todo list

จะได้ผลลัพธ์:

1 ⬜ ซื้อนม
2 ⬜ อ่านหนังสือ
3 ⬜ ออกกำลังกาย

แล้วลองทำเครื่องหมายว่าเสร็จกับลบดู:

todo done 2
todo remove 1
todo list
2 ✅ อ่านหนังสือ
3 ⬜ ออกกำลังกาย

ลองเปิดไฟล์ db.json ดูด้วย จะเห็นข้อมูลจริง ๆ ที่ถูกบันทึกไว้:

[
  {
    "id": 2,
    "text": "อ่านหนังสือ",
    "done": true
  },
  {
    "id": 3,
    "text": "ออกกำลังกาย",
    "done": false
  }
]

ตอนนี้แอปใช้งานจริงได้แล้ว ลองปิด Terminal เปิดใหม่ แล้วสั่ง todo list ข้อมูลก็ยังอยู่ครบ เหลือเพียงคำสั่ง web ซึ่งเราจะทำในบทที่ 24
