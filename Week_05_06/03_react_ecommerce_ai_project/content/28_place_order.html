Place Order

ถึงบทประกอบร่างแล้ว สามบทที่ผ่านมาเราเตรียมชิ้นส่วนไว้ครบมือ: schema กับข้อมูลตัวเลือก (บทเรียน 25), กล่องที่อยู่ (บทเรียน 26), กล่องจัดส่ง/ชำระเงิน/ใบสรุป/ยอดเงิน (บทเรียน 27) — แต่ยังไม่มีใคร render และยังไม่มี "สมอง" ที่ถือ state กับ logic การสั่งซื้อ บทนี้เราจะเติมสองอย่างสุดท้าย: ชั้น API + hook useCheckout แล้วแทนที่ stub ของหน้า Checkout และ OrderSuccess ที่ค้างมาตั้งแต่บทเรียน 09 ให้กลายเป็นหน้าจริง จบบทนี้ผู้ใช้จะช้อปครบ loop ตั้งแต่หยิบของใส่ตะกร้าจนเห็นหน้ายืนยันคำสั่งซื้อ — และแอปของคุณจะทำงานได้ครบทุกเส้นทางเป็นครั้งแรก

บทนี้เป็นบทที่กลับมาพิมพ์เองเป็นครั้งสุดท้ายก่อน chapter การทดสอบ และไฟล์ที่จะพิมพ์คือ useCheckout.ts ซึ่งมีที่ทางในลำดับงานแบบเดียวกับ cartReducer.ts เมื่อบทเรียน 22 เป๊ะ — เดี๋ยวเราจะพิสูจน์ให้ดูว่าเป๊ะจริงด้วยการไล่ลูกศรอีกครั้ง

useMutation — ทำไมไม่ใช่ query

ตั้งแต่บทเรียน 17 เราใช้ useQuery (ผ่าน query factories) มาตลอด แต่ query คือเครื่องมือของ "การอ่าน": TanStack Query จะ cache ผลลัพธ์, ยิงซ้ำให้ตอน mount, refetch ตอนกลับมาโฟกัสหน้าต่าง — พฤติกรรมพวกนี้เหมาะกับ GET ทั้งนั้น แต่ลองนึกภาพมันเกิดกับ "การสั่งซื้อ" สิ: เปิดหน้าแล้วสั่งซื้อเองอัตโนมัติ? สลับแท็บกลับมาแล้วสั่งซ้ำอีกรอบ? หายนะชัด ๆ

การเขียน (สร้าง/แก้/ลบข้อมูลฝั่งเซิร์ฟเวอร์) ต้องใช้ useMutation ซึ่งนิสัยตรงข้ามทุกข้อ: ไม่ cache, ไม่ยิงเอง, ทำงานเฉพาะเมื่อเราสั่งผ่าน mutate() เท่านั้น สิ่งที่มันให้เราคือ state ของรอบการยิงนั้น — ที่เราใช้คือ isPending (rename เป็น isPlaceOrderLoading) ซึ่งจะไหลไปเป็น prop isLoading ของปุ่ม Place Order ให้กลายเป็น spinner ระหว่างรอเซิร์ฟเวอร์ตอบ

ข้อแตกต่างอีกข้อที่สำคัญกับบทนี้เป็นพิเศษ: mutation มี onSuccess ซึ่งเป็นที่เดียวในโค้ดที่รับประกันว่าเซิร์ฟเวอร์ตอบสำเร็จแล้วจริง ๆ ทุก side effect ที่ต้องเกิด "หลังสั่งซื้อสำเร็จ" — ล้างตะกร้า, พาไปหน้าขอบคุณ — ต้องอยู่ในนั้น ไม่ใช่ที่อื่น และนั่นคือเหตุผลทั้งหมดที่ไฟล์นี้ต้องพิมพ์เอง

ผลพลอยได้ของขั้นที่ 3 คราวนี้ชัดกว่าบทเรียน 22 อีก เพราะ useCheckout คืน object ที่มีสิบเอ็ดคีย์ และมีคีย์หนึ่งที่สะกดต่างจากที่ทุกคนคาด: มันคืน billingSummary.subtotal (t เล็ก) ในขณะที่ BillingSummary จากบทที่แล้วรับ prop ชื่อ subTotal (T ใหญ่) ถ้า Agent ต้องเดา signature ของ hook มันจะเดาว่าชื่อตรงกันทั้งคู่ แล้วหน้าเพจจะพังทันทีที่ typecheck — แต่พอไฟล์จริงอยู่ในบริบท มันจะเห็นความไม่ตรงกันนี้และเขียน subTotal={subtotal} ให้ถูกตั้งแต่รอบแรก

import { useMemo, useState } from "react";
import { useMutation } from "@tanstack/react-query";
import { CheckoutAddressFormData } from "@/features/checkout/schema";
import { deliveryOptions, paymentOptions } from "@/features/checkout/consts";
import { useCart } from "@/features/cart/useCart";
import useProductRoute from "@/shared/hooks/useProductRoute";
import { placeOrder } from "@/features/checkout/api";

const useCheckout = () => {
  const [checkoutAddress, setCheckoutAddress] =
    useState<CheckoutAddressFormData | null>(null);
  const [selectedDeliveryOptionId, setSelectedDeliveryOptionId] =
    useState<number>(deliveryOptions[0].id);
  const [selectedPaymentOptionId, setSelectedPaymentOptionId] =
    useState<number>(paymentOptions[0].id);
  const { goToOrderSuccess } = useProductRoute();
  const { cart, clearCart } = useCart();

  // Helper to get selected delivery option
  const selectedDeliveryOption = useMemo(
    () =>
      deliveryOptions.find((option) => option.id === selectedDeliveryOptionId),
    [selectedDeliveryOptionId],
  );

  // Compute billing summary data
  const billingSummary = useMemo(() => {
    const subtotal = cart.items.reduce(
      (sum, item) => sum + item.price * item.quantity,
      0,
    );

    const shipping = selectedDeliveryOption?.price || 0; // Get price of the selected delivery option
    const total = subtotal + shipping;

    return { subtotal, shipping, total };
  }, [cart, selectedDeliveryOption]);

  const storeCheckoutAddress = (addressData: CheckoutAddressFormData) => {
    setCheckoutAddress(addressData);
  };

  const onSelectDeliveryOption = (optionId: number) => {
    setSelectedDeliveryOptionId(optionId);
  };

  const onSelectPaymentOption = (optionId: number) => {
    setSelectedPaymentOptionId(optionId);
  };

  const { mutate: placeOrderMutate, isPending: isPlaceOrderLoading } =
    useMutation({
      mutationFn: () => {
        if (!checkoutAddress) {
          throw new Error("Delivery Address is Missing");
        }

        return placeOrder({
          userId: 1,
          products: cart.items.map((item) => ({
            id: item.id,
            quantity: item.quantity,
          })),
          address: checkoutAddress,
        });
      },
      onSuccess: () => {
        clearCart();
        goToOrderSuccess();
      },
    });

  const onPlaceOrder = () => {
    placeOrderMutate();
  };

  return {
    checkoutAddress,
    deliveryOptions,
    paymentOptions,
    selectedDeliveryOptionId,
    selectedPaymentOptionId,
    billingSummary,
    isPlaceOrderLoading,
    storeCheckoutAddress,
    onSelectDeliveryOption,
    onSelectPaymentOption,
    onPlaceOrder,
  };
};

export default useCheckout;

เหตุผล: ลำดับของ side effect ผิดลำดับเดียว ตะกร้าหายก่อนสั่งซื้อสำเร็จ — สามบรรทัดที่ชี้ชะตาคือ onSuccess ที่เรียก clearCart() แล้ว goToOrderSuccess() ทั้งคู่อยู่ใน callback ที่ทำงานหลังเซิร์ฟเวอร์ตอบสำเร็จเท่านั้น ย้ายสองบรรทัดนี้ออกไปไว้ใน onPlaceOrder (ก่อน placeOrderMutate() หรือหลังก็ตาม) โค้ดจะยัง lint ผ่าน typecheck ผ่าน และทำงานถูกต้องทุกครั้งที่เน็ตดี — ความผิดจะโผล่เฉพาะวันที่ API ล้มเหลว ซึ่งเป็นวันที่ผู้ใช้เสียของทั้งตะกร้าโดยไม่ได้อะไรกลับมาเลย บั๊กประเภทที่ไม่มีวันเกิดในเครื่องคุณแต่เกิดกับลูกค้าคือบั๊กที่คุณต้องเคยพิมพ์โค้ดที่ถูกด้วยมือตัวเองมาก่อน ถึงจะมองออกตอนอ่านของคนอื่น

ไฟล์ยาวหน่อยแต่แบ่งเป็นห้าท่อนชัด ๆ ค่อย ๆ ไล่กัน

ท่อนที่ 1: state สามก้อนของหน้า Checkout

นี่คือ state ที่สามบทก่อนหน้า "ลอยขึ้นมารอ" — ในที่สุดก็ได้บ้านอยู่:

  • checkoutAddress — เริ่มเป็น null (ยังไม่ได้กรอก) พอฟอร์มบทเรียน 26 submit ผ่าน validation ค่าจะมาลงที่นี่ผ่าน storeCheckoutAddress
  • selectedDeliveryOptionId — ค่าเริ่มต้นคือ deliveryOptions[0].id แปลว่าหน้าเปิดมาพร้อม Standard Delivery ติ๊กอยู่แล้ว ไม่มีจังหวะไหนที่ radio ทั้งกลุ่มว่างเปล่า — UX ที่ดีของ radio คือ "มีตัวเลือกเสมอ" และการอ้าง [0].id จาก consts แปลว่าถ้าวันหน้าเราสลับลำดับ array ค่า default ก็ขยับตามเองโดยไม่ต้องแก้โค้ดตรงนี้
  • selectedPaymentOptionId — ตรรกะเดียวกัน default เป็น Cash on Delivery (ซึ่งมี id เป็น 3 ตามที่กล่อง 💡 ของบทเรียน 25 เตือนไว้ — ค่านี้ไม่ได้มาจากการนับ 1 ใหม่)

ถัดมา hook หยิบเครื่องมือเก่าสองชิ้น: goToOrderSuccess จาก useProductRoute (บทเรียน 10) ไว้พาไปหน้าสำเร็จ และ cart + clearCart จาก useCart (บทเรียน 22) — checkout อ่านตะกร้าเพื่อสร้าง order และล้างตะกร้าเมื่อสำเร็จ นี่คือครั้งแรกที่ clearCart ถูกใช้จริง หลังจากนั่งรออยู่ใน CartContextProps มาหกบทเต็ม ๆ

ท่อนที่ 2–3: ค่าที่คำนวณจาก state ด้วย useMemo

selectedDeliveryOption แปลง id ที่เลือกกลับเป็น object เต็ม ๆ (เพราะเราต้องใช้ price ของมัน) ส่วน billingSummary คือเลขสามตัวที่ BillingSummary บทที่แล้วนั่งรออยู่:

  • subtotal — reduce รวม price × quantity ของทุกชิ้นในตะกร้า สูตรเดียวกับหน้า Cart บทเรียน 24
  • shipping — ราคาของ delivery ที่เลือก (?.price || 0 กันกรณี find ไม่เจอ ซึ่งตามโครงข้อมูลจริงไม่ควรเกิด แต่ TypeScript บังคับให้เราคิดถึงมัน — ดีแล้ว)
  • total — บวกกันตรง ๆ

ทั้งสองก้อนเป็น derived state: คำนวณได้จาก state อื่นเสมอ จึงห้ามเก็บเป็น useState แยก (เดี๋ยวลืม sync) ใช้ useMemo ให้คำนวณใหม่เฉพาะเมื่อ dependencies เปลี่ยน — เลือก Express ปุ๊บ selectedDeliveryOption เปลี่ยน → billingSummary คำนวณใหม่ → ค่าส่งกับยอดรวมบนจอเด้งตามทันทีโดยไม่มีใครสั่ง

ท่อนที่ 4: mutation และ callback ที่ชี้ชะตา

ใน mutationFn มีด่านกันพลาดอยู่บรรทัดแรก: ถ้า checkoutAddress ยังเป็น null ให้ throw ทันที ตามจริงปุ่มถูก disable ไว้อยู่แล้วถ้ายังไม่กรอกที่อยู่ (เดี๋ยวเห็นในหน้าเพจ) แต่การเช็คซ้ำในชั้น logic เป็น defensive programming ที่ถูกต้อง — UI เปลี่ยนได้เสมอ และการเช็คนี้ยังทำให้ TypeScript รู้ว่าหลังบรรทัดนั้น checkoutAddress ไม่ใช่ null แน่ ๆ (type narrowing) เราจึงส่งมันเข้า payload ได้โดยไม่ต้อง ! จากนั้นก็ map ของในตะกร้าให้เหลือ { id, quantity } ตามรูปแบบที่ PlaceOrderPayload กำหนด

แล้วก็ถึง callback ที่เป็นหัวใจของ flow ทั้งบท:

onSuccess: () => {
  clearCart();
  goToOrderSuccess();
},

onSuccess ทำงานเฉพาะเมื่อเซิร์ฟเวอร์ตอบสำเร็จ: ล้างตะกร้า (dispatch CLEAR_CART → reducer คืน items: [] → write-through ของ CartProvider เขียนทับ localStorage ให้ด้วย — กลไกจากบทเรียน 22 ทำงานให้ฟรี ๆ) แล้วพาผู้ใช้ไป /order/success ลำดับนี้สำคัญนะ: ถ้าเราล้างตะกร้าก่อนยิง API แล้ว API พัง ของในตะกร้าผู้ใช้หายเกลี้ยงทั้งที่ยังไม่ได้ซื้อ — side effect ต้องเกิดหลังความสำเร็จได้รับการยืนยันเท่านั้น

และนี่คือจุดที่ต้องเน้นเป็นพิเศษเพราะมันคือแก่นของบท: onPlaceOrder เรียก placeOrderMutate() เฉย ๆ แล้วจบ — มันไม่ await อะไรทั้งนั้น เพราะ mutate ของ TanStack Query เป็นแบบ fire-and-forget ฟังก์ชันคืนค่ากลับมาทันทีก่อนที่ request จะออกจากเครื่องด้วยซ้ำ แปลว่าโค้ดบรรทัดใดก็ตามที่คุณเขียนต่อท้าย placeOrderMutate() จะทำงานก่อนผลลัพธ์กลับมาเสมอ ไม่ใช่หลัง ถ้าคุณเผลอเขียนว่า

// ผิด — อย่าเขียนแบบนี้
const onPlaceOrder = () => {
  placeOrderMutate();
  clearCart();
};

ตะกร้าจะถูกล้างทันทีที่กดปุ่ม ไม่ว่าคำสั่งซื้อจะสำเร็จหรือไม่ และในเครื่องคุณที่ dummyjson ตอบ 200 ทุกครั้ง คุณจะไม่มีวันเห็นความต่าง นี่คือเหตุผลที่ clearCart ต้องอยู่ใน onSuccess เท่านั้น และเป็นเหตุผลที่บทเรียน 32 จะเขียนเทสต์สองตัวจับข้อนี้โดยเฉพาะ: ตัวหนึ่งยืนยันว่าสั่งสำเร็จแล้วตะกร้าว่าง อีกตัวยืนยันว่าสั่งไม่สำเร็จแล้วตะกร้าต้องยังอยู่ครบ

ข้อสังเกตเรื่อง error: ถ้า API ล้มเหลว onSuccess ไม่ทำงาน ตะกร้ายังอยู่ครบ ปุ่มหยุดหมุนแล้วกลับมากดใหม่ได้ — ปลอดภัย แต่เวอร์ชันนี้ยังไม่ได้แสดงข้อความ error ให้ผู้ใช้เห็น (useMutation มี error ให้หยิบอยู่แล้ว แค่ hook เรายังไม่ return ออกไป) ถือเป็นแบบฝึกหัดต่อยอดที่ดีมาก: ลอง return error ออกไปแล้วแสดง ErrorMessage เหนือปุ่มดูเมื่อจบคอร์ส

ท่อนที่ 5: return — สัญญาของ hook

ก้อน return คือ "API ของหน้า Checkout" ทั้งหมดในที่เดียว: ข้อมูล (ที่อยู่, รายการตัวเลือก, id ที่เลือก, ยอดเงิน, สถานะ loading) กับ handlers (เก็บที่อยู่, เลือกจัดส่ง/จ่ายเงิน, สั่งซื้อ) สังเกตว่า hook ส่งต่อ deliveryOptions / paymentOptions จาก consts ออกไปด้วย — หน้าเพจเลยไม่ต้อง import consts เอง ทุกอย่างเกี่ยวกับ checkout ไหลผ่านประตูบานเดียว ซึ่งจะทำให้ทดสอบง่ายมากในบทเรียน 32 (renderHook ตัวเดียว เห็นทุกอย่าง) และสังเกตด้วยว่าสิ่งที่ไม่ได้ถูก return คือ placeOrderMutate ตัวจริง — คนนอกได้แค่ onPlaceOrder ที่ไม่รับพารามิเตอร์ ปิดโอกาสที่ใครจะยิง mutation ด้วย payload ของตัวเอง

💡 ของที่ควรถูกทักในรีวิว — สามข้อในไฟล์ที่คุณเพิ่งพิมพ์เอง: ข้อแรกคือ comment สามบรรทัดที่คุณเพิ่งพิมพ์ตามไปด้วย — // Helper to get selected delivery option, // Compute billing summary data และ // Get price of the selected delivery option ทั้งสามอันเขียนซ้ำสิ่งที่ชื่อตัวแปรบรรทัดถัดไปบอกอยู่แล้วทุกตัวอักษร (selectedDeliveryOption, billingSummary, shipping = selectedDeliveryOption?.price) ซึ่งเป็น comment ประเภทที่ checklist ของบทเรียน 22 ปฏิเสธไว้ตรง ๆ ว่า "ไม่มี comment ที่เขียนซ้ำสิ่งที่โค้ดบอกอยู่แล้ว" — เราให้คุณพิมพ์ตามเพื่อให้ diff ตรงกับโค้ดอ้างอิง แต่ในรีวิวจริงสามบรรทัดนี้ควรถูกลบ (comment ที่ควรอยู่คือ comment ที่อธิบายการตัดสินใจ แบบใน catch ของ saveCartState ไม่ใช่ comment ที่แปลโค้ดเป็นภาษาอังกฤษ) ข้อสองคือ selectedDeliveryOption?.price || 0 ที่ควรเป็น ?? 0 — วันนี้ผลลัพธ์เหมือนกันเพราะไม่มีตัวเลือกไหนราคา 0 แต่วันที่ธุรกิจเพิ่ม "Free Shipping" ราคา 0 เข้ามา || จะมองว่า 0 เป็นค่าเท็จแล้วแทนที่ด้วย 0 (บังเอิญได้ผลถูก) — และวันที่ค่านั้นเป็นส่วนลดติดลบหรือค่าที่ 0 มีความหมายต่างจาก "ไม่มีค่า" มันจะไม่บังเอิญถูกอีกต่อไป ?? ถามว่า "เป็น null/undefined ไหม" ซึ่งตรงกับเจตนาจริงของบรรทัดนี้ ข้อสามคือชื่อ: hook คืน billingSummary.subtotal ตัว t เล็ก ส่วน BillingSummary ที่เพิ่งสร้างเมื่อบทที่แล้วรับ prop ชื่อ subTotal ตัว T ใหญ่ ผลคือหน้าเพจต้องเขียน subTotal={subtotal} ซึ่งเป็นบรรทัดที่อ่านแล้วสะดุดทุกครั้ง สองไฟล์ในฟีเจอร์เดียวกัน สองธรรมเนียมการสะกดคำเดียวกัน — เป็นรูปแบบเดิมที่คอร์สนี้เจอมาตลอด: สิ่งที่ไม่มีเครื่องมือบังคับจะเพี้ยนไปทีละไฟล์ และคราวนี้มันมีราคาจริงเพิ่มอีกชั้น เพราะเทสต์ในบทเรียน 32 จะ assert result.current.checkout.billingSummary.subtotal ตามชื่อฝั่ง hook ส่วนบทเรียน 33 อ่านตัวเลขจากหน้าจอที่ BillingSummary วาด — ใครแก้ชื่อฝั่งใดฝั่งหนึ่งให้ตรงกันทีหลัง จะทำเทสต์พังโดยไม่รู้ตัว

src/features/checkout/api.ts — ยิงคำสั่งซื้อขึ้นเซิร์ฟเวอร์:

import { apiClient } from "@/shared/api/client";
import { CheckoutAddressFormData } from "@/features/checkout/schema";

export interface PlaceOrderPayload {
  userId: number;
  products: { id: number; quantity: number }[];
  address: CheckoutAddressFormData;
}

export interface PlaceOrderResponse {
  id: number;
}

export async function placeOrder(
  payload: PlaceOrderPayload,
): Promise<PlaceOrderResponse> {
  const { data } = await apiClient.post<PlaceOrderResponse>(
    "/carts/add",
    payload,
  );
  return data;
}

โครงเหมือน features/products/api.ts จากบทเรียน 16 เป๊ะ — ฟังก์ชัน async บาง ๆ ที่เรียก apiClient (axios instance จากบทเรียน 15) แล้วคืน data ที่ type แล้ว จุดที่ต่างคือคราวนี้เป็น .post() ครั้งแรกของคอร์ส: เราส่งข้อมูลขึ้นไป ไม่ใช่ดึงลงมา และนี่คือไฟล์ที่บทเรียน 15 ทำนายไว้ว่าจะเป็นผู้ใช้รายที่สองของ client.ts ต่อจาก products

  • PlaceOrderPayload — รวมสามส่วน: userId (hardcode เป็น 1 เพราะเราไม่มีระบบ login), รายการสินค้าแบบย่อ (เอาแค่ id + quantity พอ เซิร์ฟเวอร์รู้ราคาเองอยู่แล้ว — อย่าเชื่อราคาที่ client ส่งมา!) และ address ที่ reuse type CheckoutAddressFormData จาก schema ตรง ๆ — schema เดียว ไหลตั้งแต่ฟอร์มยัน payload ซึ่งเป็นเหตุผลทั้งหมดที่บทเรียน 25 ให้คุณพิมพ์ schema เองแทนที่จะปล่อยให้ Agent ประกาศ interface ซ้ำ
  • PlaceOrderResponse — dummyjson ตอบกลับ cart ก้อนใหญ่ แต่เราประกาศ type เฉพาะ field ที่ใช้จริงคือ id การ type ให้แคบที่สุดเท่าที่ใช้เป็นนิสัยที่ดี: response เปลี่ยนโครงเมื่อไร โค้ดเราพังเฉพาะจุดที่พึ่งมันจริง ๆ
  • error ไม่ต้องจัดการที่นี่ — interceptor ใน client.ts แปลงทุกความล้มเหลวเป็น ApiError ให้แล้วตั้งแต่บทเรียน 15 ชั้นนี้แค่ปล่อยให้ reject ไหลขึ้นไป
💡 ของที่ควรถูกทักในรีวิว: POST /carts/add ของ dummyjson เป็น mock endpoint — มันรับ payload, ตอบกลับมาเป็น cart จำลองพร้อม id ที่ปั้นขึ้นมา แต่ไม่ได้บันทึกอะไรลงฐานข้อมูลจริง (echo back) dummyjson ไม่มีระบบ order ให้เราใช้ ซึ่งสำหรับการเรียนแบบนี้คือข้อดี: กดสั่งซื้อกี่รอบก็ได้โดยไม่ทิ้งขยะไว้ที่ไหน แต่ในเชิงรีวิวมีสองข้อที่ต้องพูด ข้อแรก: ชื่อฟังก์ชันคือ placeOrder แต่ URL คือ /carts/add — คนที่มาอ่านโค้ดนี้ในอีกหกเดือนจะสับสนว่าตกลงมันสร้าง order หรือสร้าง cart กันแน่ comment หนึ่งบรรทัดว่า "dummyjson ไม่มี endpoint order ใช้ /carts/add แทน" คือ comment ประเภทที่ควรมี (อธิบายการตัดสินใจ ไม่ใช่แปลโค้ด) และไฟล์นี้ไม่มี ข้อสอง: field address ใน payload เป็นของที่ dummyjson ไม่รู้จักด้วยซ้ำ มันรับแล้วเมิน เราใส่ไว้เพราะโค้ดฝั่งเราควรเขียนเหมือน API จริง — วันที่ย้ายไป backend จริง สิ่งที่เปลี่ยนคือ URL ไม่ใช่โครงโค้ด นั่นเป็นการตัดสินใจที่ถูก แต่ก็เป็นการตัดสินใจที่ไม่มีใครนอกจากคนเขียนรู้ ถ้าไม่เขียนไว้

หน้านี้แทบไม่มี logic เลย — และนั่นแหละคือความสำเร็จ มันคือ "wiring diagram" ล้วน ๆ:

  • CategoryBanner name="Checkout" — และนี่คือการ reuse ครั้งที่สามในสามครั้งที่บทเรียน 20 ทำนายไว้ตอนสร้างมัน (หน้า Product บทเรียน 21, หน้า Cart บทเรียน 24 และหน้านี้) คอมโพเนนต์ที่ถูกออกแบบให้รับแค่ name?: string | null ตัวเดียวจึงย้ายบ้านไปไหนก็ได้ — ถ้าวันนั้นมันรับ Category ทั้งก้อนตามที่ Agent เสนอ วันนี้หน้า Checkout จะไม่มีอะไรส่งให้มันเลย
  • layout สองคอลัมน์ — flex gap-20 แบ่งซ้าย 60% (addressBox: กล่องที่อยู่ → จัดส่ง → ชำระเงิน เรียงลงมา) ขวา 40% (summaryBox: ใบสรุป + ยอดเงิน) ตาม SCSS module สองบรรทัดข้างบน
  • ทุก prop มาจาก useCheckout ก้อนเดียว — destructure ทีเดียวแล้วแจกจ่าย รวมถึงท่า destructure ซ้อน billingSummary: { subtotal, shipping, total } ที่แกะเลขสามตัวออกมาในบรรทัดเดียว
  • isDisabled={!checkoutAddress} — ด่านกันพลาดฝั่ง UI ที่สัญญาไว้: ตราบใดที่ยังไม่ submit ที่อยู่ ปุ่ม Place Order จะกดไม่ได้ พอที่อยู่มาปุ๊บปุ่มติดเอง เพราะ prop คำนวณใหม่ทุก render
  • SummaryOrder ไม่รับ prop อะไรเลย — ตามที่ออกแบบไว้บทที่แล้ว มันอ่านตะกร้าจาก context เอง

สังเกตว่าหน้านี้ไม่มี guard กรณีตะกร้าว่าง — เข้าตรง ๆ ที่ /checkout ทั้งที่ตะกร้าโล่งก็จะเห็นใบสรุปว่าง ๆ กับยอด $5.99 (ค่าส่งอย่างเดียว) ซึ่งใช้งานได้แต่ไม่สวยนัก โปรเจกต์ต้นฉบับเลือกเก็บหน้าให้เรียบง่ายไว้ก่อน ถ้าอยากฝึกเพิ่ม ลอง redirect ไปหน้า Cart เมื่อ cart.items.length === 0 เป็นการบ้านหลังจบคอร์สได้ — แต่ให้ทำหลังบทเรียน 33 เพราะ integration test ของหน้านี้เตรียมตะกร้าไว้ล่วงหน้าแล้วค่อยเปิดหน้า การใส่ redirect โดยไม่แก้เทสต์จะทำให้มันพัง

💡 ของที่ควรถูกทักในรีวิว: ลำดับของชื่อในก้อน destructure สิบเอ็ดตัวไม่ได้เรียงตามอะไรเลย — storeCheckoutAddress (handler) มาก่อน checkoutAddress (ค่า), แล้ว selectedPaymentOptionId โผล่ก่อน selectedDeliveryOptionId ถึงห้าบรรทัด ทั้งที่ทั้งคู่เป็นของคู่กัน และ onPlaceOrder ไปแทรกอยู่ตรงกลางระหว่างสอง id ที่ควรอยู่ติดกัน ผลคือเวลาอ่านหา prop สักตัวต้องกวาดตาทั้งก้อน แทนที่จะกระโดดไปกลุ่มที่ใช่ — เทียบกับก้อน return ใน useCheckout ที่เรียงเป็นระเบียบ (ค่าทั้งหมดก่อน แล้วค่อย handlers) จะเห็นว่าปลายทางทำหายไปเอง ทางแก้คือ copy ลำดับจากก้อน return ของ hook มาใช้ตรง ๆ ซึ่งได้ผลพลอยได้ว่าเวลามีคนเพิ่มคีย์ใหม่ใน hook คุณจะรู้ทันทีว่าต้องเติมตรงไหนในหน้าเพจ เราสั่งตามต้นฉบับใน prompt เพื่อให้ diff ตรง แต่นี่คือ comment หนึ่งบรรทัดที่แก้ง่ายที่สุดในบทนี้ ข้อพ่วง: บรรทัดว่างระหว่าง </div> ของ SummaryOrder กับ <BillingSummary> เป็นบรรทัดว่างเดียวใน JSX ทั้งไฟล์ ทั้งที่คู่กล่องฝั่งซ้ายสามใบไม่มีคั่นเลย — Prettier ไม่ยุ่งกับบรรทัดว่างใน JSX (มันเก็บไว้ตามที่คนเขียนใส่) ดังนั้นความไม่สม่ำเสมอนี้จะอยู่ตรงนั้นตลอดไปจนกว่าจะมีคนสังเกต ซึ่งเป็นเหตุผลที่ prompt ของเราต้องสั่งเรื่องบรรทัดว่างไว้ด้วย ไม่งั้น diff จะไม่มีวันตรง

ทดสอบ happy path ทั้งเส้น

นี่คือช่วงเวลาที่รอมาตั้งแต่บทเรียน 01 — flow ช้อปปิ้งครบ loop ครั้งแรกของแอป รัน npm run dev แล้วไล่ตามนี้ทีละข้อ:

  1. เปิดหน้า Product ของสินค้าสักชิ้น กด Add to Cart (จะเพิ่มหลายชิ้นก็ยิ่งดี)
  2. เข้าหน้า Cart ตรวจของ แล้วกดปุ่ม Checkout
  3. เห็นหน้าใหม่เอี่ยม: แบนเนอร์ Checkout, กล่องซ้ายสามใบ, ใบสรุปขวาที่มีสินค้าของเราครบ และปุ่ม Place Order ที่ยังกดไม่ได้
  4. ลองกด Submit ฟอร์มทั้งที่ว่าง → ข้อความแดง "Address is required" ฯลฯ โผล่ครบสามช่อง — schema ที่คุณพิมพ์เองในบทเรียน 25 ทำงานแล้ว ลองพิมพ์ email มั่ว ๆ ดูด้วยว่าได้ "Enter a valid email address"
  5. กรอกให้ครบแล้ว Submit → ที่อยู่ขึ้นไปแสดงในกล่อง และปุ่ม Place Order ติดสว่างพร้อมกด
  6. สลับ delivery เป็น Express → Shipping ฝั่งขวาเด้งเป็น $15.99 และ Total ขยับตามทันที เลือก payment ตามใจ แล้วสังเกตว่า radio สองกลุ่มไม่รบกวนกัน (นี่คือข้อพิสูจน์ว่า name ของสองกลุ่มต่างกันจริงตามที่ checklist บทเรียน 27 ให้ตรวจ)
  7. กด Place Order → ปุ่มกลายเป็น spinner แวบหนึ่ง → เด้งไปหน้า Order Success ไอคอนติ๊กถูกเขียว
  8. ดู Header: ตัวเลขบนไอคอนตะกร้าเป็น 0 แล้ว — CLEAR_CART ที่คุณเขียนไว้ใน reducer เมื่อบทเรียน 22 เพิ่งถูกใช้จริงเป็นครั้งแรก

แล้วลองเส้นทางที่ไม่มีใครอยากให้เกิดดูสักครั้ง เพราะมันคือทั้งเหตุผลที่บทนี้ให้คุณพิมพ์ hook เอง: เปิด DevTools แท็บ Network เลือก throttling เป็น Offline แล้วกด Place Order — ปุ่มจะหมุนแล้วหยุด ไม่มีอะไรเกิดขึ้น ไม่เด้งไปไหน และของในตะกร้าต้องยังอยู่ครบทุกชิ้น (เช็คที่ badge บน Header) นั่นคือ onSuccess ที่ไม่ถูกเรียกทำงานถูกต้อง ถ้าตะกร้าว่างเปล่าหลังการทดลองนี้ แปลว่ามี clearCart อยู่ผิดที่ กลับไปอ่านข้อ AI มักพลาดตรงนี้ อีกรอบ

🤔 อธิบายให้ได้

  1. ย้าย clearCart() จาก onSuccess ออกมาไว้ใน onPlaceOrder ต่อจาก placeOrderMutate() แล้วทำสองการทดลอง: (ก) กดสั่งซื้อตามปกติ (ข) เปิด Network เป็น Offline แล้วกดสั่งซื้อ — ตอบว่าผู้ใช้เห็นอะไรในแต่ละกรณี แล้วอธิบายว่าทำไมการทดลอง (ก) ถึงดูเหมือนทุกอย่างปกติดี และบอกด้วยว่าถ้าคุณเป็นคนรีวิว pull request นี้โดยไม่ได้รันโค้ด คุณจะจับมันได้จากบรรทัดไหน (เสร็จแล้วย้ายกลับ)
  2. useCheckout return onPlaceOrder ออกไปแทนที่จะ return placeOrderMutate ตรง ๆ ทั้งที่สองตัวนี้ทำงานเหมือนกันเมื่อเรียกโดยไม่มีอาร์กิวเมนต์ — อธิบายว่าเราได้อะไรจากการห่ออีกชั้น แล้วตอบต่อว่าถ้าหน้าเพจได้ placeOrderMutate ไปตรง ๆ มันจะทำอะไรได้บ้างที่เราไม่อยากให้ทำ (เปิดเอกสารของ useMutation ดูว่า mutate รับพารามิเตอร์อะไรได้บ้าง)
  3. ไล่เส้นทางของตัวเลข "ค่าส่ง $15.99" ให้ครบตั้งแต่ผู้ใช้คลิก radio จนตัวเลขเปลี่ยนบนจอ — ผ่านกี่ไฟล์ ไฟล์อะไรบ้าง ใครถือ state ใครคำนวณ ใครแค่แสดง จากนั้นตอบคำถามที่ยากกว่า: ถ้าเราอยากให้ราคาค่าส่งไปโผล่ที่แถวของตัวเลือกด้วย (ตามที่กล่อง 💡 ในบทเรียน 27 ทักไว้) ต้องแก้กี่ไฟล์ ไฟล์ไหนบ้าง และมีไฟล์ไหนที่ไม่ต้องแก้เลยทั้งที่เกี่ยวข้องกับราคา
💡 Checkpoint: แอปของคุณขายของได้แล้ว — ครบ loop ตั้งแต่เลือกของยันยืนยันคำสั่งซื้อ เดินตามข้อ "ทดสอบ happy path ทั้งเส้น" ให้ครบทั้งแปดข้อบวกการทดลอง Offline อีกหนึ่งครั้ง แล้วปิดท้ายด้วยหลักฐานสามชิ้น: (1) DevTools → Application → Local Storage → key ecommerce-cart:v1 ต้องกลายเป็น {"items":[]} หลังสั่งซื้อสำเร็จ — write-through จากบทเรียน 22 เขียนทับตะกร้าที่ล้างแล้วให้เรียบร้อย refresh กี่ครั้งตะกร้าก็ยังว่าง (2) แท็บ Network ระหว่างกด Place Order ต้องเห็น POST https://dummyjson.com/carts/add หนึ่งรายการเท่านั้น พร้อม response จำลองที่มี id กลับมา — ถ้าเห็นสองรายการ แปลว่ามีใครเรียก mutate ซ้ำ (3) npm run lint และ npm run typecheck ผ่านทั้งคู่ แล้ว commit chapter ชำระเงินจบบริบูรณ์ และฟีเจอร์ทั้งหมดของแอปจบแล้ว — จากนี้ไปเราจะไม่เพิ่มหน้าใหม่อีกเลย บทหน้าเป็นบทที่ต่างจากทุกบทที่ผ่านมา: เราจะจงใจให้ Agent เขียนผิด แล้วฝึกอ่านให้ออกว่าผิดตรงไหน ก่อนจะเข้า chapter การทดสอบที่จะทำให้ทุกอย่างที่คุณสร้างมา 28 บทพิสูจน์ได้ว่าทำงานถูกต้อง โดยเริ่มจาก schema ที่คุณพิมพ์เองในบทเรียน 25 และ hook ที่คุณพิมพ์เองวันนี้ — ถ้าข้อ "อธิบายให้ได้" ยังตอบไม่ได้ อย่าเพิ่งไปบทถัดไป
