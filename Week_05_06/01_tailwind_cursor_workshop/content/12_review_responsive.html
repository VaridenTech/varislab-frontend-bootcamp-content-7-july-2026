แก้ TODO ข้อ 1 — ความกว้างตายตัว

เปิด DevTools device toolbar เลือก 390px — scrollbar แนวนอนโผล่ทันที ไล่หาต้นตอ: DashboardLayout.tsx มี w-[1440px] mx-auto — Agent อยากได้ "เนื้อหากว้างสุด 1440 จัดกลาง" แต่เขียนเป็นความกว้างตายตัว: จอแคบกว่า 1440 ก็ยังยืนยันจะกว้าง 1440 จนล้น

- <div className="w-[1440px] mx-auto">
+ <div className="mx-auto w-full max-w-[1440px] px-4 md:px-6">

max-w-[1440px] = ไม่เกิน 1440 แต่ยอมหดตามจอ, w-full = ใช้พื้นที่ที่มี, px-4 md:px-6 = กันเนื้อหาชนขอบจอ — สามตัวนี้รวมกันคือ pattern "กรอบเนื้อหา" จากบทเรียน 04

แก้เองก็ได้ หรือสั่ง Agent แก้ก็ได้ — prompt สั่งแก้ที่ดีชี้ไฟล์ ชี้ปัญหา ชี้วิธีแก้ สั้น ๆ

ใน src/app/layouts/DashboardLayout.tsx เปลี่ยน "w-[1440px] mx-auto" เป็น
"mx-auto w-full max-w-[1440px] px-4 md:px-6" — ตอนนี้จอเล็กเกิด horizontal scroll
เพราะความกว้างตายตัว

จากนั้นอ่าน diff เพื่อยืนยันว่า Agent แก้เฉพาะจุดที่สั่ง ถ้ามีการจัดรูปแบบทั้งไฟล์หรือเปลี่ยนส่วนอื่นจนเทียบได้ยาก ให้ reject แล้วระบุขอบเขตของคำสั่งให้แคบลง

Checklist ตรวจ Responsive

นี่คือเลนส์แรกจากสามเลนส์ ใช้ไล่ทุกหน้า ทุกครั้งที่รับงานจาก Agent — ต่อไปบทอื่นจะเรียกว่า "checklist บทเรียน 12"

  • ทดสอบที่ 390 / 768 / 1024 / 1440 ทุกครั้ง — สี่จุดนี้จับปัญหาได้เกือบหมด
  • ค้นทั้งโปรเจกต์หา w-[ และ h-[ ที่เป็นเลขตายตัว — ถามทีละตัวว่า "จอแคบกว่านี้จะเกิดอะไร" ถ้าคำตอบคือ "ล้น" ให้เปลี่ยนเป็น max-w-* + w-full (ค่าตายตัวที่ตั้งใจเช่นความกว้าง sidebar เก็บไว้ได้)
  • ทุก grid ต้องมีลำดับคอลัมน์จากมือถือขึ้นไป (grid-cols-1 md:… xl:…)
  • ตารางข้อมูลต้องอยู่ใน overflow-x-auto
  • รูปมี object-cover + สัดส่วน (aspect-*) — กันรูปบี้เมื่อกล่องเปลี่ยนขนาด
  • ข้อความยาวที่คุมไม่ได้ (ชื่อจาก data) มี truncate หรือ line-clamp-*
  • ไม่มี horizontal scroll ที่ความกว้างใดเลย (ลากขอบจอช้า ๆ ทั้งช่วง)
  • ของที่ hidden บนมือถือ ต้องมีทางเข้าถึงอื่นเสมอ

ข้อสุดท้ายตรงกับปัญหาสำคัญอีกจุดของหน้านี้

แก้ TODO ข้อ 2 — sidebar ต้องมี drawer บนมือถือ

ตอนนี้ sidebar เป็น hidden lg:flex — บนมือถือเมนูทั้งแอปหายไป วิธีมาตรฐานคือ drawer: ปุ่ม hamburger บน navbar เปิด sidebar ที่เลื่อนเข้ามาจากซ้าย พร้อมฉากหลังทึบ นี่คืองานที่มี state ข้ามสอง component เราจึงยก state ขึ้นไปไว้ที่ layout

// src/app/layouts/DashboardLayout.tsx
import { useState } from "react";
import { Outlet } from "react-router";
import Sidebar from "../../components/Sidebar";
import Navbar from "../../components/Navbar";

export default function DashboardLayout() {
  const [sidebarOpen, setSidebarOpen] = useState(false);

  return (
    <div className="min-h-screen bg-surface">
      <Sidebar open={sidebarOpen} onClose={() => setSidebarOpen(false)} />
      <div className="lg:pl-[280px]">
        <div className="mx-auto w-full max-w-[1440px] px-4 md:px-6">
          <Navbar onMenuClick={() => setSidebarOpen(true)} />
          <main className="pb-10">
            <Outlet />
          </main>
        </div>
      </div>
    </div>
  );
}

ฝั่ง Sidebar: เลิกใช้ hidden แล้วใช้การเลื่อนออกนอกจอแทน — จอเล็กซ่อนด้วย -translate-x-full เปิดเมื่อ open, จอใหญ่ตรึงไว้เสมอด้วย lg:translate-x-0

// src/components/Sidebar.tsx (โครงใหม่)
interface SidebarProps {
  open: boolean;
  onClose: () => void;
}

export default function Sidebar({ open, onClose }: SidebarProps) {
  return (
    <>
      {open && (
        <button
          aria-label="ปิดเมนู"
          onClick={onClose}
          className="fixed inset-0 z-30 bg-black/50 lg:hidden"
        />
      )}
      <aside
        className={`fixed inset-y-0 left-0 z-40 flex w-[260px] flex-col bg-surface px-6 py-8
          transition-transform lg:translate-x-0
          ${open ? "translate-x-0" : "-translate-x-full"}`}
      >
        {/* เนื้อหา sidebar เดิมทั้งหมด */}
      </aside>
    </>
  );
}

และปุ่ม hamburger ใน Navbar — แสดงเฉพาะจอเล็ก (lg:hidden)

// src/components/Navbar.tsx (เพิ่ม)
<button
  onClick={onMenuClick}
  aria-label="เปิดเมนู"
  className="rounded-lg p-2 text-muted lg:hidden"
>
  <svg className="size-6" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" aria-hidden="true">

รายละเอียดที่อย่าลืม: กดลิงก์ในเมนูแล้วเรียก onClose ด้วย (ไม่งั้น drawer ค้างบังหน้าใหม่) และปุ่มทั้งสองมี aria-label เพราะเป็นปุ่มไอคอนล้วน — เดี๋ยวเจอหลักการเต็ม ๆ ในบทถัดไป

แก้ TODO ข้อ 5/8 — ตารางล้นจอ

ตาราง Projects (หน้า Dashboard) และตารางทั้งสองของหน้า Tables ล้นจอมือถือ — ท่ามาตรฐานของตารางข้อมูลคือยอมให้ scroll แนวนอนภายในการ์ดของมันเอง

<div className="overflow-x-auto">
  <table className="w-full min-w-[640px] text-left">
    {/* ... */}
  </table>
</div>

ใช้โครงนี้ครอบทั้งสามตาราง โดย min-w ป้องกันไม่ให้คอลัมน์แคบจนอ่านไม่ได้ ผู้ใช้มือถือจะเลื่อนดูคอลัมน์ด้านข้างภายในการ์ดได้โดยที่ทั้งหน้าไม่ขยับ วิธีนี้เหมาะกับตารางสำหรับอ่านข้อมูล ส่วนการเปลี่ยนแต่ละแถวเป็นการ์ดจะใช้โครงสร้างมากกว่าและไม่จำเป็นสำหรับงานนี้
