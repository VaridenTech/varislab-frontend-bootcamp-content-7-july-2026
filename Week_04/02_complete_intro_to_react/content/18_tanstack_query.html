TanStack Query

มาสร้างหน้า past orders กัน สร้างไฟล์ใหม่ชื่อ past.lazy.jsx หาก Vite server ของคุณกำลังทำงานอยู่ มันจะสร้าง stub ให้โดยอัตโนมัติ! ประสบการณ์สำหรับนักพัฒนาที่เจ๋งมากจาก TanStack

ทีนี้มาติดตั้งเครื่องมือชิ้นถัดไปของเรา TanStack (React) Query

npm i @tanstack/react-query@5.59.13
npm i -D @tanstack/react-query-devtools@5.59.13 @tanstack/eslint-plugin-query@5.59.7

React Query ทำให้การเรียก async API ง่ายขึ้นมาก ง่ายกว่า/ดีกว่ามากจนฉันแทบไม่ใช้ useEffect อีกแล้วเหมือนที่เราใช้กับรายการ pizzas หรือ pizza of the day โค้ดสะอาดกว่า อ่านง่ายกว่า cache ดีกว่า และมี bug น้อยกว่า สรุปคือมันดีกว่า คุณควรรู้วิธีใช้ effects แต่จากนี้ไปก็ใช้ TanStack Query สำหรับ API calls

มาเริ่มกัน เริ่มจากเพิ่ม ESLint config ของมันเข้า config ของเรา มันมี rules ที่มีประโยชน์ ใน eslint.config.mjs

// at top
import pluginQuery from "@tanstack/eslint-plugin-query";

// under reactPlugin.configs.flat["jsx-runtime"]
...pluginQuery.configs["flat/recommended"],

นี่จะเพิ่ม rules ของ ESLint สองสามข้อที่เฉพาะกับ React Query ฉันพบว่ามันมีประโยชน์ และที่สำคัญกว่านั้นคือไม่น่ารำคาญ จึงมักเพิ่มมันทุกครั้ง

เพิ่ม dev tools ด้วยเหมือนที่ทำกับ router ใน src/routes/__root.jsx:

// at top
import { ReactQueryDevtools } from "@tanstack/react-query-devtools";

// under router dev tools
<ReactQueryDevtools />

สุดท้ายเราต้องเพิ่ม QueryClient ใน App.jsx เพิ่ม:

// Add imports
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";

// Create a property under the router
const queryClient = new QueryClient()

// Add the provider to the app
<QueryClientProvider client={queryClient}>
  <RouterProvider router={router} />
</QueryClientProvider>

ตอนนี้มุมขวาล่างของหน้าต่างจะเห็น 🏝️ คลิกมันเพื่อเปิด dev tools เรายังไม่ได้ใช้ dev tools จึงยังว่างอยู่

react-query ทำให้การทำงานกับ APIs ง่ายมากและอ่านง่าย คุณแค่อ่าน hook แล้วมันจะให้สถานะ isLoading หรือ data กลับมา เมื่อ data กลับมา มันจะ refresh component ด้วย data นั้น เริ่มจากเขียน fetch call ที่ง่ายมาก สร้างโฟลเดอร์ api ใน src แล้วสร้าง getPastOrders.js ใส่:

export default async function getPastOrders(page) {
  const response = await fetch(`/api/past-orders?page=${page}`);
  const data = await response.json();
  return data;
}

เป็น request ที่ง่ายมากไปยัง API ที่ return data แค่นั้น!

ทีนี้ไปสร้าง past.lazy.jsx

import { useState } from "react";
import { useQuery } from "@tanstack/react-query";
import { createLazyFileRoute } from "@tanstack/react-router";
import getPastOrders from "../api/getPastOrders";

export const Route = createLazyFileRoute("/past")({
  component: PastOrdersRoute,
});

function PastOrdersRoute() {
  const [page, setPage] = useState(1);
  const { isLoading, data } = useQuery({
    queryKey: ["past-orders", page],
    queryFn: () => getPastOrders(page),
    staleTime: 30000,
  });
  if (isLoading) {
    return (
      <div className="past-orders">
        <h2>LOADING …</h2>
      </div>
    );
  }
  return (
    <div className="past-orders">
      <table>
        <thead>
          <tr>
            <td>ID</td>
            <td>Date</td>
            <td>Time</td>
          </tr>
        </thead>
        <tbody>
          {data.map((order) => (
            <tr key={order.order_id}>
              <td>{order.order_id}</td>
              <td>{order.date}</td>
              <td>{order.time}</td>
            </tr>
          ))}
        </tbody>
      </table>
      <div className="pages">
        <button disabled={page <= 1} onClick={() => setPage(page - 1)}>
          Previous
        </button>
        <div>{page}</div>
        <button disabled={data.length < 10} onClick={() => setPage(page + 1)}>
          Next
        </button>
      </div>
    </div>
  );
}
  • เราใช้ hooks เพื่อติดตามหน้าปัจจุบัน
  • เราใช้ useQuery hook เพื่อเรียก API และส่ง queryFn ที่บอกวิธีไป fetch data
  • เราส่ง keys ซึ่งทำหน้าที่เป็น cache keys เราใช้ past-orders เป็น key แต่จะเป็น unique key อะไรก็ได้สำหรับหน้านี้ แล้วใส่ page ต่อท้าย สิ่งที่เจ๋งคือ แม้ครั้งแรกเราจะ request page 1 แต่ครั้งที่สองที่ request page 1 มันจะเห็นว่ามีใน cache แล้วและไม่ request ใหม่ เจ๋งแค่ไหน? เมื่อก่อนต้องใช้ logic เยอะมากเพื่อจัดการเรื่องนี้ แต่ตอนนี้ React Query ทำให้คุณเอง
  • ตอนนี้เปิด dev tools คุณเห็นหน้าทั้งหมดกำลังโหลดอยู่ เจ๋งใช่ไหม?
  • ลองเอา page ออกจาก query key มันจะเตือนคุณ นี่คือ ESLint config ที่เราดึงมาจาก Tanstack Query เพราะเราใช้ page ใน request จึงต้องใช้มันเป็น caching key หากคุณพึ่งพา variable เพื่อ request มันควรเป็นส่วนหนึ่งของ caching key
  • เราให้ staleTime เป็น 30 วินาที (30,000 milliseconds) สิ่งนี้ทำให้คนที่ใช้หน้า browse ไปมาเล็กน้อยได้โดยไม่ยิง API มากเกินไป แต่หน้าก็ไม่ stale เกินไป หากไม่ใส่ staleTime มันจะ refetch ทุกครั้ง

แค่นั้น! React Query ทั้งใช้ง่ายและยืดหยุ่นมากสำหรับจัดการปัญหาที่ยาก มันเป็นหนึ่งใน libraries React ที่ฉันชอบที่สุด
