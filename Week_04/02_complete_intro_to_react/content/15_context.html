การแชร์ State ด้วย Context

Context

Context

มาสร้าง cart indicator ที่มุมขวาบนของหน้า สร้างไฟล์ชื่อ Header.jsx แล้วใส่สิ่งนี้ลงไป

export default function Header() {
  return (
    <nav>
      <h1 className="logo">Pizza app</h1>
      <div className="nav-cart">
        🛒<span className="nav-cart-number">5</span>
      </div>
    </nav>
  );
}

ตอนนี้มาใช้มันใน App.jsx

// at top
import Header from "./Header";

// replace .logo
<Header />

ตอนนี้มันจะแสดง 5 เสมอ แต่เราต้องการให้ตัวเลขใน .nav-cart-number สะท้อนจำนวน items ใน cart ของเรา จะทำอย่างไร? เราอาจย้าย cart ทั้งหมดและ hooks ของมันไป App.jsx แล้วส่งมันเข้า Header และ Order ในแอปเล็กเท่านี้ นั่นอาจเป็นทางเลือกที่ถูกต้อง แต่มาดูอีกวิธีหนึ่ง คือ context

Context มีไว้สำหรับ app-level state นี่คือ state ที่มีอยู่กับทั้ง app ของคุณ ผู้ใช้ที่เข้าสู่ระบบอยู่ในขณะนี้เป็นตัวอย่างที่ดี คุณคงไม่ต้องการให้ผู้ใช้มีอยู่แค่ระดับ page เพราะเมื่อคุณไปหน้าอื่น state ทั้งหมดของหน้าก่อนหน้าจะถูกทำลาย คุณย่อมต้องการให้ข้อมูลผู้ใช้นั้นคงอยู่ระหว่าง pages ดังนั้น context จึงเหมาะกับเรื่องนั้น Context ให้คุณเก็บ global state สำหรับ app ของคุณ เราจะใช้กับ shopping cart ซึ่งก็สมเหตุสมผลมาก: หากผู้ใช้ไปหน้าอื่น คุณย่อมต้องการเก็บ shopping cart ไว้ระหว่าง loads นี่เป็นวิธีคิดที่ดีว่าควรใช้ context เมื่อไร: หากคุณต้องการให้ state นั้นคงอยู่ระหว่าง pages

ดังนั้นมาทำให้มันทำงาน สร้างไฟล์ชื่อ contexts.jsx มันไม่ใช่ component ดังนั้นฉันมักไม่ใช้ตัวพิมพ์ใหญ่ React docs ใช้ตัวพิมพ์ใหญ่ คุณเลือกได้

import { createContext } from "react";

export const CartContext = createContext([[], function () {}]);

แค่นั้นเอง ง่ายมาก มันชื่อ contexts ที่มี s เพราะถ้าคุณมี contexts อื่น (เช่น UserContext) คุณก็ใส่ทั้งหมดในไฟล์เดียวกันได้ [[], function () {}] ไม่จำเป็นโดยเคร่งครัด; มันคือ default value ที่ context จะใช้หากไม่มี context provider อยู่ (ซึ่งไม่ควรเกิดขึ้น) โดยจริง ๆ แล้วมันมีประโยชน์กับ TypeScript types เท่านั้น — type ที่คุณให้ที่นี่คือสิ่งที่ TypeScript จะใช้ validate ในทางทฤษฎีมันอาจมีประโยชน์กับ testing ได้ด้วย เหตุผลที่ value แปลกคือมันเป็น React hook: array ที่ value แรกเป็น array (เช่น cart ของเรา) และ value ที่สองเป็น function (function setCart)

โอเค มาใส่มันใน App.jsx

// at the top
import { StrictMode, useState } from "react"; // need useState
import { CartContext } from "./contexts";

// replace App
const App = () => {
  const cartHook = useState([]);
  return (
    <StrictMode>
      <CartContext.Provider value={cartHook}>
        <div>
          <Header />
          <Order />
          <PizzaOfTheDay />
        </div>
      </CartContext.Provider>
    </StrictMode>
  );
};

เรากำลังสร้าง hook (สังเกตว่าเราไม่ได้ destructuring) แล้วส่งมันเข้าเป็น value Provider ทำให้ context นั้นใช้ได้เฉพาะภายใน children ของมัน ในทางทฤษฎีคุณมีหลาย providers จาก context เดียวกันได้ แต่พูดตามจริงฉันไม่เคยมีเหตุผลต้องทำ

เอาล่ะ มาทำให้ Order.jsx ใช้งานได้

// at top
import { useState, useEffect, useContext } from "react"; // need useContext
import { CartContext } from "./contexts";

const [cart, setCart] = useContext(CartContext); // change cart hook to use context

แค่นั้น! อย่างอื่นทำงานเหมือนเดิม กรณีของเราเราเพียงใช้ hook โดยตรงจาก context แต่คุณใส่อะไรก็ได้ใน context; ไม่จำเป็นต้องเป็น hook

มาใช้มันใน Header ของเราด้วย

// at top
import { useContext } from "react";
import { CartContext } from "./contexts";

// top of function
const [cart] = useContext(CartContext);

// replace span number
🛒<span className="nav-cart-number">{cart.length}</span>

แค่นั้น! เราไม่ต้องใช้ function setCart ดังนั้นจึงไม่ import มัน นอกนั้นก็ดูปกติดี!

นี่คือ context มันมีประโยชน์มากกับเรื่องแบบนี้ที่เราไม่อยาก “prop drill” state ของเราไปทั่ว ตอนนี้ cart ถูกเก็บไว้ระดับ App เราจะเขียน <Header cart={cart} /> แล้วจบก็ได้ และใน use case นี้ฉันคงทำแบบนั้น แต่ลองนึกภาพว่า Header ซ้อนลึกมากและคุณต้องส่ง state นั้นลงผ่าน children components จำนวนมาก — มันจะยุ่งอย่างรวดเร็ว นี่คือสิ่งที่เรียกว่า prop drilling data flow ที่ชัดเจนของ React เป็น feature ไม่ใช่ bug มันทำให้ที่มาของ data และที่ที่ data ไปอ่านง่ายมาก แต่มันทำให้โค้ดยาวโดยไม่เกิดประโยชน์ และ context คือ escape hatch สำหรับเรื่องนั้น โดยทั่วไป ใช้ context อย่างประหยัด และใช้เฉพาะเมื่อการใช้ props ตามวิธีปกติไม่สะดวกจริง ๆ
