Testing User Interaction

มาทดสอบหน้า Contact กัน มันมี user interaction ที่ดีและเรียบง่าย ผู้ใช้ใส่ข้อมูล เรา submit ไป API แล้วแสดง submitted header ให้พวกเขา มาทดสอบทั้งหมดนั้นกัน

ก่อนอื่นเราต้องมี helper function, vitest-fetch-mock เราไม่ได้ ต้อง ใช้มัน เพราะ Vitest ทำได้ แต่มันช่วยให้เขียนสั้นลงมาก

npm i -D vitest-fetch-mock@^0.4.5

และตอนนี้มาเขียน code กัน สร้างไฟล์ contact.lazy.test.tsx ใน src/__tests__

import { render } from "@testing-library/react";
import { expect, test, vi } from "vitest";
import createFetchMock from "vitest-fetch-mock";
import { QueryClientProvider, QueryClient } from "@tanstack/react-query";
import { ContactRoute } from "../routes/contact.lazy";

const queryClient = new QueryClient();

const fetchMocker = createFetchMock(vi);
fetchMocker.enableMocks();

test("can submit contact form", async () => {
  fetchMocker.mockResponse(JSON.stringify({ status: "ok" }));
  const screen = render(
    <QueryClientProvider client={queryClient}>
      <ContactRoute />
    </QueryClientProvider>
  );

  const nameInput = screen.getByPlaceholderText("Name") as HTMLInputElement;
  const emailInput = screen.getByPlaceholderText("Email") as HTMLInputElement;
  const msgTextArea = screen.getByPlaceholderText(
    "Message"
  ) as HTMLTextAreaElement;

  const testData = {
    name: "Brian",
    email: "test@example.com",
    message: "This is a test message",
  };

  nameInput.value = testData.name;
  emailInput.value = testData.email;
  msgTextArea.value = testData.message;

  const btn = screen.getByRole("button");

  btn.click();

  const h3 = await screen.findByRole("heading", { level: 3 });

  expect(h3.innerText).toContain("Submitted");

  const requests = fetchMocker.requests();
  expect(requests.length).toBe(1);
  // requests() เก็บ URL ที่ resolve เป็น absolute แล้ว (http://localhost:3000/api/contact)
  // เทียบเฉพาะ pathname จึงตรงประเด็นกว่าและไม่ผูกกับ base URL ของ happy-dom
  expect(new URL(requests[0].url).pathname).toBe("/api/contact");
  expect(fetchMocker).toHaveBeenCalledWith("/api/contact", {
    body: JSON.stringify(testData),
    headers: {
      "Content-Type": "application/json",
    },
    method: "POST",
  });
});

หมายเหตุผู้แปล: ต้นฉบับเขียนบรรทัดนั้นว่า expect(requests[0].url).toBe("/api/contact") ซึ่งจะไม่ผ่าน เพราะ fetchMocker.requests() เก็บ URL ที่ resolve เป็น absolute เทียบกับ base URL ของ happy-dom แล้ว (http://localhost:3000/api/contact) ไม่ใช่ string ที่เราส่งเข้า fetch() · เราจึงเทียบเฉพาะ pathname ซึ่งนอกจากจะผ่านแล้วยังตรงกับสิ่งที่เราสนใจจริง ๆ มากกว่าด้วย คือ "ยิงไปที่ endpoint ถูกตัวไหม" ไม่ใช่ "host คืออะไร" · สังเกตว่าบรรทัด toHaveBeenCalledWith("/api/contact", …) ถัดลงไปไม่ต้องแก้ เพราะมันดู argument ที่เราส่งเข้า fetch() จริง ๆ ไม่ใช่ค่าที่ resolve แล้ว

ก่อนที่เทสต์นี้จะทำงานได้ ต้องแก้อีกที่หนึ่ง กลับไปที่ src/routes/contact.lazy.tsx แล้วเติม export ให้ component

// เดิม: function ContactRoute() {
export function ContactRoute() {
  // …
}

ทำไมไม่ใช้ Route.options.component

ต้นฉบับที่เป็น JavaScript เขียน <Route.options.component /> คือหยิบ component ออกมาจาก route object ซึ่งทำได้เพราะ JavaScript ไม่ได้ตรวจอะไรเลย แต่ใน TypeScript วิธีนี้จะไม่ผ่าน เพราะ TanStack Router ประกาศ component ไว้เป็น optional (component?: RouteComponent) — route หนึ่ง ๆ จะไม่มี component ก็ได้ TypeScript จึงไม่ยอมให้เราเอาสิ่งที่อาจเป็น undefined มาใช้เป็น JSX tag

เราจะดันทุรังด้วย Route.options.component! ก็ได้ แต่การ export component ออกมาตรง ๆ ดีกว่าในทุกทาง: type ถูกต้องโดยไม่ต้องหลอกใคร อ่านแล้วรู้ทันทีว่ากำลังเทสต์อะไร และที่สำคัญที่สุดคือมันตรงกับหลักที่เราพูดไว้ตลอดว่า ให้เทสต์สิ่งที่ผู้ใช้เห็น — สิ่งที่เราสนใจคือหน้า Contact ทำงานถูกไหม ไม่ใช่ว่า router เก็บ component ไว้ที่ property ไหน

อีกจุดคือ as HTMLInputElement และ as HTMLTextAreaElement ด้วยเหตุผลเดียวกับ as HTMLImageElement ในบทที่แล้ว: เราต้องเข้าถึง .value ซึ่งไม่มีใน HTMLElement กลาง ๆ สังเกตว่า <textarea> ต้องใช้ HTMLTextAreaElement ไม่ใช่ HTMLInputElement — ถ้าสลับกัน TypeScript จับได้ทันที

  • เราต้อง wrap React Query ด้วย query provider เสมอ ไม่เช่นนั้นมันไม่ทำงาน กรณีนี้เราให้มันเป็นส่วนหนึ่งของ initial render ได้เลย
  • คุณสามารถ pre-populate React Query ด้วย cache ที่ใช้ได้ แล้วทดสอบว่า app ใช้มันถูกต้องหรือไม่ เป็นวิธี test ที่ใช้ได้อย่างสมบูรณ์ ในกรณีนี้เราอยากให้แน่ใจว่า endpoint ของเราถูก post อย่างถูกต้อง (เพื่อให้แน่ใจว่าเราบันทึก contact) ดังนั้นจึงไม่ทำเช่นนั้น
  • vi คือ spy library ของ Vitest คล้าย sinon
  • vitest-fetch-mock เป็นเพียง layer ที่สะดวกบน vi เราใช้ vi โดยตรงก็ได้
  • สังเกตว่าเราแค่เรียก click ได้เลย และมันทำทุกอย่างอย่างที่คาด Cool ใช่ไหม?
  • จากนั้นเราทำ assertions ว่า h3 ของเราแสดงถูกต้อง API ถูกเรียกครั้งเดียว และถูก submit ไปยัง API ของเราในแบบที่เราคาด นี่อาจดูเน้น implementation เล็กน้อย ซึ่งก็ใช่ แต่โดยทั่วไปใน app ใหญ่กว่านี้เราน่าจะมี library สำหรับเรียก API และทดสอบแยกต่างหาก แต่ที่นี่เราไม่มี เราจึงรวมมันทั้งหมดไว้ด้วยกัน
