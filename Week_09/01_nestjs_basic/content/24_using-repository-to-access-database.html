ใช้ Repository เข้าถึงฐานข้อมูล

เมื่อจบบทนี้ คุณจะสามารถ

  • ฉีด Repository<Coffee> เข้า service ด้วย @InjectRepository()
  • แปลง CRUD ทั้ง 5 method จาก array ในหน่วยความจำมาเป็นการเรียกฐานข้อมูลจริง
  • เลือกใช้ create, save, preload, find, remove ให้ถูกกับสถานการณ์

Repository คือประตูสู่ตารางของ entity

ทุก entity ที่ลงทะเบียนผ่าน forFeature() จะมี repository ประจำตัวให้ใช้ query ตารางนั้นโดยเฉพาะ เราขอมันมาใช้ผ่าน constructor injection ด้วย decorator @InjectRepository(Coffee) — pattern เดียวกับที่ฉีด provider อื่น ๆ ของ Nest จึงทดสอบและ mock ได้ง่ายในบทเรื่อง testing

import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { CreateCoffeeDto } from './dto/create-coffee.dto';
import { UpdateCoffeeDto } from './dto/update-coffee.dto';
import { Coffee } from './entities/coffee.entity';

@Injectable()
export class CoffeesService {
  constructor(
    @InjectRepository(Coffee)
    private readonly coffeeRepository: Repository<Coffee>,
  ) {}

  findAll() {
    return this.coffeeRepository.find();
  }

  async findOne(id: string) {
    const coffee = await this.coffeeRepository.findOne({ where: { id: +id } });
    if (!coffee) {
      throw new NotFoundException(`Coffee #${id} not found`);
    }
    return coffee;
  }

  create(createCoffeeDto: CreateCoffeeDto) {
    const coffee = this.coffeeRepository.create(createCoffeeDto);
    return this.coffeeRepository.save(coffee);
  }

  async update(id: string, updateCoffeeDto: UpdateCoffeeDto) {
    const coffee = await this.coffeeRepository.preload({
      id: +id,
      ...updateCoffeeDto,
    });
    if (!coffee) {
      throw new NotFoundException(`Coffee #${id} not found`);
    }
    return this.coffeeRepository.save(coffee);
  }

  async remove(id: string) {
    const coffee = await this.findOne(id);
    return this.coffeeRepository.remove(coffee);
  }
}

method ไหนทำอะไร

Methodทำอะไรจุดที่ควรสังเกต
find() / findOne()ดึงหลายแถว / แถวเดียวตามเงื่อนไข wherefindOne คืน null เมื่อไม่พบ — เราจึงต้อง throw NotFoundException เอง
create()สร้าง instance ของ entity ในหน่วยความจำยังไม่แตะฐานข้อมูล ต้องตามด้วย save()
preload()โหลดแถวเดิมจาก id แล้ว merge ค่าใหม่ทับให้คืน undefined ถ้า id ไม่มีจริง — เหมาะกับ flow ของ update
save()insert หรือ update ลงฐานข้อมูลจริงใช้ได้ทั้งกับของใหม่จาก create และของเดิมจาก preload
remove()ลบ entity ที่โหลดมาแล้วออกจากตารางเรา reuse findOne ก่อน เพื่อให้ id ที่ไม่มีจริงได้ 404 เหมือนกัน

สังเกตว่า id รับเข้ามาเป็น string จาก route parameter แล้วแปลงเป็นตัวเลขด้วย +id ก่อนใช้ query — ในบทที่ 56 เราจะเห็นวิธีที่สะอาดกว่าด้วย ParseIntPipe และ custom pipes

ย้าย CoffeesService ลงฐานข้อมูลจริง

แทนที่ array และ logic เดิมทั้งหมดใน CoffeesService ด้วยโค้ดด้านบน แล้วทดสอบ CRUD ครบวงจรผ่าน Insomnia

POST http://localhost:3000/coffees { "name": "Shipwreck Roast", "brand": "Buddy Brew", "flavors": ["chocolate", "vanilla"] } GET http://localhost:3000/coffees GET http://localhost:3000/coffees/1 PATCH http://localhost:3000/coffees/1 { "name": "Updated Roast" } DELETE http://localhost:3000/coffees/1

ผลลัพธ์ที่คาดหวัง

ทุก endpoint ทำงานเหมือนเดิมจากมุมมองของ client แต่ข้อมูลถูกเก็บถาวรแล้ว — restart แอปแล้วข้อมูลไม่หาย และเห็นแถวจริงในตาราง coffee ผ่าน pgAdmin ส่วน id ที่ไม่มีจริงได้ response 404 พร้อมข้อความ Coffee #id not found

ตรวจว่าใช้งานได้จริง

  • POST แล้วเห็นแถวใหม่ในตาราง coffee ผ่าน pgAdmin หรือ SELECT * FROM coffee;
  • restart แอป (Ctrl+C แล้ว npm run start:dev ใหม่) แล้ว GET ยังเห็นข้อมูลเดิม
  • GET /coffees/9999 ได้ status 404 พร้อม message ที่อ่านรู้เรื่อง
  • PATCH ด้วย id จริงได้ค่าใหม่กลับมา และ PATCH ด้วย id ปลอมได้ 404 (พิสูจน์ว่า preload ทำงาน)
