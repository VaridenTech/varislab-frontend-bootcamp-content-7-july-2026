ข้อควรระวังกับ Async Middleware

Middleware ในโลกจริงแทบทุกตัวเป็น Async เพราะมันต้องคุยกับ Database หรือยิงไปหา Service อื่น และของพวกนั้นจะพังสักวันแน่นอน บทนี้เราจะเก็บกับดักที่เจอบ่อยที่สุดให้ครบก่อนจะเข้าสู่เรื่อง Database ในบทหน้า

เบื่อ try/catch ไหม — Middleware Factory ช่วยได้

ตัวผมเองเขียน try/catch ในทุก Handler จนเบื่อ เพราะข้างใน Handler ผมคุยกับ Database ตลอด ถ้าไม่ Catch ไว้ Server ผมตาย วิธีแก้ที่ผมใช้จริงคือเขียนตัวห่อ ที่รับ Middleware เข้าไปหนึ่งตัว แล้วคืน Middleware ที่จัดการ Error ให้เสร็จสรรพออกมา

import type { RequestHandler } from 'express'

export const asyncHandler =
  (fn: RequestHandler): RequestHandler =>
  (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next)
  }

อ่านจากข้างในออกมา — มันเอาผลลัพธ์ของ Middleware ของเราไปห่อด้วย Promise.resolve() เพื่อให้แน่ใจว่าได้ Promise ออกมาเสมอ แล้วต่อ .catch(next) ท้ายไว้ ซึ่งแปลว่าถ้ามันพังเมื่อไร Error จะถูกส่งเข้า next() ทันที และตามที่เราเรียนไปในบทที่ 10 อะไรก็ตามที่ส่งเข้า next() ถือเป็น Error และจะวิ่งไปหา Error Handler

เวลาใช้ก็แค่ห่อทับเข้าไป

const loadHabit: RequestHandler = async (req, res, next) => {
  await someAsyncOperation(req.params.id)   // คุยกับ Database
  next()
}

router.get('/:id', asyncHandler(loadHabit), getHabit)

ผลคือเราเขียน Logic การจัดการ Errorครั้งเดียว แทนที่จะก๊อปวาง try/catch ไปทุกไฟล์ นี่คือโค้ดที่ผมยกมาจากโปรเจกต์จริงของผมเองเลย

คอร์สนี้เราจะยังไม่ใส่ตัวนี้ลงโปรเจกต์ เพราะ Express 5 มีของแบบนี้ให้ในตัวแล้ว (เดี๋ยวอธิบายในกับดักข้อ 3) แต่คุณควรรู้จักมันไว้ เพราะโค้ดที่คุณจะเจอในโลกจริงเต็มไปด้วยรูปแบบนี้

กับดักที่ 1: ลืมเรียก next()

ย้ำอีกครั้งเพราะเจอบ่อยที่สุด ถ้าไม่ตอบ ไม่โยน Error และไม่เรียก next() Request จะค้าง ไม่มี Error ไม่มีอะไรทั้งนั้น Express แค่นั่งรอเฉย ๆ ซึ่งหาเจอยากกว่า Error จริง ๆ เสียอีก

// แย่ — Request ค้างตลอดกาล
const badMiddleware = (req, res, next) => {
  console.log('Processing...')
}

// ดี
const goodMiddleware = (req, res, next) => {
  console.log('Processing...')
  next()
}

กับดักที่ 2: เรียงลำดับ Middleware ผิด

ตัวอย่างคลาสสิกคือพยายามอ่าน req.body ก่อนที่ express.json() จะทำงาน

// แย่ — body ยังไม่ถูก Parse
app.use((req, res, next) => {
  console.log(req.body.name)  // undefined
  next()
})
app.use(express.json())       // สายไปแล้ว

// ดี — Parser มาก่อน
app.use(express.json())
app.use((req, res, next) => {
  console.log(req.body.name)  // ใช้ได้
  next()
})

ประเด็นทั้งหมดของ express.json() คือมันเป็นคนสร้าง req.body ขึ้นมา ถ้ามันยังไม่ได้รัน req.body ก็ไม่มีอยู่จริงสำหรับ Route นั้น ไม่ใช่ว่าว่างเปล่า แต่คือไม่มีเลย ลำดับสำคัญเสมอ

กับดักที่ 3: ไม่จัดการ Error ของ Async

สมัยก่อนข้อนี้เป็นปัญหาใหญ่มาก ตอนนั้นเรายังไม่มี async/await ไม่มีแม้กระทั่ง Promise ทุกอย่างเป็น Callback หมด การลืมจัดการ Error จึงเกิดขึ้นง่ายมาก

// แย่ — Unhandled Promise Rejection
const badAsyncMiddleware = async (req, res, next) => {
  await someAsyncOperation()  // ถ้าตัวนี้โยน Error ไม่มีใครรับ
  next()
}

// ดี — ห่อด้วย try/catch แล้วส่งต่อให้ Error Handler
const goodAsyncMiddleware = async (req, res, next) => {
  try {
    await someAsyncOperation()
    next()
  } catch (error) {
    next(error)
  }
}

สาเหตุคือ Express 4 ไม่รู้จัก Promise มันเรียก Middleware ของเราแบบธรรมดา ไม่ได้ await ผลลัพธ์ พอ Async Function โยน Error สิ่งที่เกิดคือ Promise ถูก Reject โดยไม่มีใครรับ กลายเป็น Unhandled Promise Rejection ที่ Express ไม่รู้เรื่องด้วย Request ก็ค้าง และใน Node.js รุ่นใหม่ Unhandled Rejection ทำให้โปรเซสตายไปเลยด้วย นั่นคือเหตุผลที่เกิดรูปแบบ asyncHandler ขึ้นมา

Express 5 แก้เรื่องนี้ให้แล้ว ถ้า Handler หรือ Middleware ของเราเป็น Async แล้ว Promise ถูก Reject Express 5 จะรับมันไปส่งให้ Error Handler ให้อัตโนมัติ เหมือนกับที่เราเรียก next(error) เอง ทุกวันนี้ข้อนี้จึงเป็นปัญหาน้อยลงมาก ส่วนหนึ่งเพราะ async/await เป็นสิ่งที่เราคุ้นมือกันอยู่แล้วจากงานฝั่ง Frontend

แต่ก็อย่าเพิ่งชะล่าใจสองเรื่อง — Express 5 จับได้เฉพาะ Promise ที่ Reject ออกมาจากตัว Middleware โดยตรง ถ้าคุณยิง Promise ทิ้งไว้เฉย ๆ โดยไม่ await มันก็ยังหลุดอยู่ดี และโค้ดเก่าที่ยังอยู่บน Express 4 กติกาเดิมยังใช้อยู่ทุกข้อ

กับดักที่ 4: แตะ Response หลังส่ง Header ไปแล้ว

ที่ผ่านมาผมบอกว่า "ตอบซ้ำไม่ได้" ซึ่งพูดให้ตรงเป๊ะกว่านั้นคือ ส่ง Header ซ้ำไม่ได้ ซึ่งความหมายเดียวกัน เพราะทุก Response เริ่มต้นด้วยการส่ง Header ไปก่อน

// แย่ทุกบรรทัด
const problematicMiddleware = (req, res, next) => {
  res.json({ message: 'First response' })

  setTimeout(() => {
    res.json({ message: 'Second response' })  // Error: headers already sent
  }, 1000)

  next()  // ตอบไปแล้วยังเรียก next() อีก
}

ทางแก้คือกฎข้อเดิม — ถ้าจะตอบ ให้ใส่ return นำหน้าและจบตรงนั้น การตอบ Response แปลว่างานเสร็จแล้ว ก็ไม่ควรมีโค้ดอะไรวิ่งต่ออีก

return res.status(400).json({ error: 'Validation failed' })

โค้ดยังวิ่งต่อได้จริงหลังตอบไปแล้ว

ลองพิสูจน์ด้วยตัวเองสักครั้ง ใส่ console.log ต่อท้าย Handler ของ /health แล้วยิงดู

app.get('/health', (req, res) => {
  res.status(200).json({ status: 'OK' })
  console.log('ยังทำงานอยู่นะ')   // บรรทัดนี้รันจริง
})

คุณจะเห็น Log โผล่ใน Terminal ทั้งที่ Client ได้ Response ไปเรียบร้อยแล้ว ซึ่งพิสูจน์ว่า Node.js ทำได้ในทางเทคนิค แต่ผมไม่คิดว่ามันเป็นแนวปฏิบัติที่ดี

เหตุผลคือคุณจะไม่มีทางรู้เลยว่างานนั้นเสร็จเมื่อไร เสร็จแล้วจะทำอะไรต่อ ถ้ามันพังจะ Retry อย่างไร และกี่ครั้ง เพราะทุกอย่างอยู่ในขอบเขตของ Handler ที่ปิดไปแล้ว คุณเข้าไม่ถึงมันอีก สิ่งที่คุณได้มาคือ Race Condition ล้วน ๆ

สิ่งเดียวที่ผมยอมทำแบบนี้คืองานเล็ก ๆ ที่ไม่กินทรัพยากร ไม่ Block อะไร และถ้าพลาดไปบ้างผมก็ไม่เดือดร้อน เช่นยิง Analytics หรือส่ง Log ขึ้น Dashboard เท่านั้น

เมื่อไรที่ "ตอบก่อน ทำต่อทีหลัง" ถึงจำเป็นจริง

มีกรณีที่รูปแบบนี้ถูกต้องจริง ๆ คือ Webhook ธรรมชาติของ Webhook คือ Service ปลายทางอย่าง Stripe หรือ Google ยิงเข้ามาหาเรา แล้วเขาคาดหวังให้เราตอบ 200 กลับไปเร็วที่สุด ถ้าเราช้าเกิน SLA ของเขา เขาอาจยิงซ้ำเหตุการณ์เดิม อาจคิดเงินเราเพิ่ม หรือถึงขั้นตัดบริการ

แล้วถ้างานที่ต้องทำเพื่อตอบสนอง Webhook นั้นใช้เวลานานล่ะ จะรอให้เสร็จก่อนค่อยตอบ Stripe เหรอ — ทำไม่ได้ ทางแก้แบบดั้งเดิมคือสร้าง Job ยัดเข้า Queue แล้วตอบกลับทันที จากนั้นมี Service อีกตัวคอยอ่านงานจาก Queue ไปทำ ส่วนแพลตฟอร์มยุคใหม่อย่าง Vercel มีคำสั่งชื่อ waitUntil ที่แปลว่า "อย่าเพิ่งปิด Serverless Function นี้ รอจนงาน Async ตัวนี้เสร็จ แต่ระหว่างนี้ตอบ Response กลับไปได้เลย" คิดง่าย ๆ ว่ามันคือ Queue แบบไม่ต้องมี Queue

ทั้งหมดนี้เป็นคนละเรื่องกับสิ่งที่เราทำใน Express และผมไม่แนะนำให้ทำแบบนั้นบน Express ด้วยเหตุผลข้างบน แต่ยกมาให้เห็นว่ามีคนคิดเรื่องนี้จริงจังจนสร้างเครื่องมือมารองรับ

สรุปแนวปฏิบัติของ Middleware

  • ลำดับสำคัญ — ความปลอดภัยก่อน Parse ข้อมูลตาม Business Logic ท้ายสุด
  • จัดการ Error เสมอ — Async ต้องมี try/catch หรือมีตัวห่อที่ทำแทน
  • ตอบหรือไม่ก็ next() — อย่าปล่อยให้ Request ค้าง และอย่าทำทั้งสองอย่าง
  • ทำอย่างเดียวให้ดี — Middleware หนึ่งตัวควรมีหน้าที่เดียว
  • ใช้ Factory — Higher-Order Function ทำให้ Middleware รับ Option และเอาไปใช้ซ้ำได้
  • เขียน Test — Middleware ที่เราเขียนเองควรมี Unit Test (เราจะทำกันในบทที่ 29)
  • บอกให้ชัดว่าแก้อะไร — ถ้า Middleware ไปแตะ req หรือ res ควรมีคอมเมนต์บอกไว้
  • ระวังเรื่องความเร็ว — Global Middleware รันทุก Request อย่าเอางานหนักไปใส่

เท่านี้เรื่อง Middleware ก็จบครบแล้ว คุณเข้าใจกลไกที่เป็นหัวใจของ Express และของ Framework ฝั่ง Node.js เกือบทั้งหมด บทหน้าเราจะเปลี่ยนเรื่องไปทางฝั่งข้อมูลกันบ้าง เริ่มจาก Database Schema และ ORM เพื่อให้ Habit Tracker ของเรามีข้อมูลจริง ๆ เก็บสักที
