Export Types และ Schema Best Practices

Schema ที่เราเขียนไปในบทที่แล้วมีข้อมูลครบทุกอย่างอยู่ในตัวมันแล้ว ทั้งชื่อ Field ชนิดข้อมูล ตัวไหนบังคับ ตัวไหนมีค่า Default บทนี้เราจะดึงข้อมูลชุดนั้นออกมาใช้ต่อ โดยไม่ต้องเขียนอะไรซ้ำอีกเลย

Export TypeScript Type จากตาราง

เติมโค้ดนี้ไว้ท้ายไฟล์ src/db/schema.ts

export type User = typeof users.$inferSelect
export type NewUser = typeof users.$inferInsert

export type Habit = typeof habits.$inferSelect
export type NewHabit = typeof habits.$inferInsert

export type Entry = typeof entries.$inferSelect
export type NewEntry = typeof entries.$inferInsert

export type Tag = typeof tags.$inferSelect
export type NewTag = typeof tags.$inferInsert

export type HabitTag = typeof habitTags.$inferSelect
export type NewHabitTag = typeof habitTags.$inferInsert

คำว่า $inferSelect กับ $inferInsert เป็นของที่ Drizzle แถมมาให้กับทุกตาราง เราใช้ typeof ครอบเข้าไปเพื่อดึง Type ของมันออกมาแล้วตั้งชื่อใหม่

สองตัวนี้ต่างกันชัดเจนและเป็นจุดที่ควรเข้าใจ

  • $inferSelect คือหน้าตาของข้อมูลตอนที่ดึงออกมาจากฐานข้อมูล มีครบทุก Field ทั้ง id, createdAt, updatedAt เพราะของพวกนั้นถูกสร้างขึ้นแล้ว
  • $inferInsert คือหน้าตาของข้อมูลตอนที่จะยัดลงฐานข้อมูล Field ที่มี Default หรือถูกสุ่มให้จะกลายเป็น Optional หมด เหลือแค่ที่เราต้องกรอกจริง ๆ ซึ่งของ User คือ email, username และ password

พอมี Type สองชุดนี้ Function ที่คุยกับฐานข้อมูลจะเขียนได้สวยมาก

const createUser = async (userData: NewUser): Promise<User> => {
  const [user] = await db.insert(users).values(userData).returning()
  return user
}

อ่านแล้วเข้าใจทันทีว่ารับของที่ยังไม่ถูกสร้าง แล้วคืนของที่ถูกสร้างแล้ว โดยที่เราไม่ต้องเขียน Interface ไหนขึ้นมาเองสักบรรทัด และถ้าวันหนึ่งเราเพิ่มคอลัมน์ใน Schema Type ทั้งหมดนี้จะอัปเดตตามให้เอง

สร้าง Zod Schema จากตารางด้วย drizzle-zod

Type ช่วยเราได้แค่ตอนเขียนโค้ด แต่ตอน Runtime มันหายไปหมด ถ้าอยากตรวจค่าจริงเราต้องใช้ Zod เหมือนที่ทำใน env.ts และใน Validation Middleware

คำถามคือ ในเมื่อ Schema ของ Drizzle รู้กฎทุกข้ออยู่แล้ว ทำไมเราต้องมานั่งเขียน Zod Schema ซ้ำอีกรอบ คำตอบคือไม่ต้อง มี Package ชื่อ drizzle-zod ที่แปลงให้

npm install drizzle-zod

เพิ่ม Import ไว้บนหัวไฟล์ schema.ts

import { createInsertSchema, createSelectSchema } from 'drizzle-zod'

แล้วเติมส่วนนี้ก่อนหน้าส่วน Export Type

export const insertUserSchema = createInsertSchema(users)
export const selectUserSchema = createSelectSchema(users)

export const insertHabitSchema = createInsertSchema(habits)
export const selectHabitSchema = createSelectSchema(habits)

export const insertEntrySchema = createInsertSchema(entries)
export const selectEntrySchema = createSelectSchema(entries)

export const insertTagSchema = createInsertSchema(tags)
export const selectTagSchema = createSelectSchema(tags)

export const insertHabitTagSchema = createInsertSchema(habitTags)
export const selectHabitTagSchema = createSelectSchema(habitTags)

สิ่งที่ได้คือ Zod Schema เต็มตัวที่สะท้อนกฎของฐานข้อมูลเป๊ะ ๆ ทั้งเรื่องความยาว varchar ตัวไหนบังคับ และตัวไหนเป็น Optional โดยเราไม่ได้เขียนกฎซ้ำเลยสักข้อ

เหตุผลที่ต้องมีสองตัวก็เหมือนเหตุผลของ Type สองชุด insertUserSchema ใช้ตรวจของที่กำลังจะสร้าง ส่วน selectUserSchema ใช้ตรวจของที่ดึงออกมาแล้วในกรณีที่เราอยากมั่นใจตอน Runtime ว่าข้อมูลหน้าตาถูกต้อง

เอาไปใช้กับ Validation Middleware

คำถามที่ตามมาทันทีคือ Schema พวกนี้ใช้แทน Zod Schema ที่เราเขียนมือใน src/middleware/validation.ts ตอนบทที่ 12 ได้ไหม คำตอบคือได้ และควรใช้ด้วย

import { insertHabitSchema } from '../db/schema.ts'

router.post('/', validateBody(insertHabitSchema), createHabit)

ตอนบทที่ 12 เราเขียน Schema เองเพราะยังไม่มีฐานข้อมูล ตอนนี้มีแล้วก็ดึงมาใช้เลยดีกว่า เพราะกฎจะไม่มีวันหลุดจากกัน

แต่มีข้อยกเว้นสำคัญที่ต้องรู้ Schema จากฐานข้อมูลใช้ตรงกับ Input ของผู้ใช้ได้ก็ต่อเมื่อทุก Field ที่บังคับ ผู้ใช้เป็นคนส่งมาทั้งหมด

ลองนึกถึงตอนสร้าง Habit ฐานข้อมูลบังคับให้มี userId แต่ผู้ใช้ไม่ได้ส่งค่านั้นมาใน Body เราต่างหากที่เป็นคนเติมให้จาก Token ถ้าเราเอา insertHabitSchema ไปตรวจ Body ตรง ๆ Request ที่ถูกต้องทุกอันจะถูกปฏิเสธหมด เพราะขาด userId

ทางแก้คือหยิบเฉพาะส่วนที่ต้องการ Zod มี Method อย่าง .omit(), .pick() และ .partial() ให้ตัดแต่งได้ หรือจะ .extend() เพื่อเพิ่มกฎที่ฐานข้อมูลไม่รู้จักก็ได้ เช่นบังคับความยาวขั้นต่ำของรหัสผ่าน ซึ่งเป็นกฎของแอปเรา ไม่ใช่กฎของฐานข้อมูล

import { z } from 'zod'

export const createUserSchema = insertUserSchema
  .omit({ id: true, createdAt: true, updatedAt: true })
  .extend({
    password: z.string().min(8, 'Password must be at least 8 characters'),
  })

เทคนิคนี้จะกลับมามีบทบาทตอนคุยเรื่อง Validation ของระบบสมัครสมาชิก (บทที่ 22 และ 30) — ในคอร์สเราเลือกใช้ Schema ที่ได้มาฟรีไปก่อน

Best Practices ของการออกแบบ Schema

ก่อนจะไปต่อ ขอรวบเจ็ดข้อที่ผมยึดทุกครั้งที่ออกแบบ Schema

  • ใส่ Timestamp เสมอ — createdAt กับ updatedAt มีประโยชน์มากกว่าที่คุณคิด และวันที่คุณต้องการมันคือวันที่มันไม่มี
  • บังคับกฎที่ระดับ Database — ใช้ unique, notNull และ Foreign Key ให้เป็นประโยชน์ เป็นชั้นที่ไม่มีใครข้ามได้
  • ทำ Index ให้คอลัมน์ที่ Query บ่อย — โดยเฉพาะ Foreign Key ทุกตัว
  • ตั้งชื่อให้สื่อความหมาย — ผมทนไม่ได้กับ Schema ที่ตั้งชื่อคอลัมน์ว่า tg แล้วไม่มีใครในทีมบอกได้ว่ามันคืออะไร
  • เผื่ออนาคตไว้ด้วย — ถ้ารู้ว่ามีฟีเจอร์อะไรอยู่ใน Roadmap ให้คิดตั้งแต่ตอนนี้ว่ามันจะกระทบ Schema อย่างไร เพราะ Migration กำลังมาแน่
  • Normalize แต่พอดี — Normalize คือการเลือกว่าจะวาง Key ไว้ฝั่งไหนแล้ว Query จากฝั่งนั้นตลอด ซึ่งเป็นเป้าหมายที่ดี แต่อย่ายึดจนถึงขั้นต้องรื้อ Logic หลังบ้านสามเดือนเพียงเพื่อความสวยงาม ถ้าเพิ่ม Key อีกตัวแล้วจบก็เพิ่มไปเถอะ
  • ใช้ Enum เมื่อค่าที่เป็นไปได้มีจำกัด — Enum ก็คือ "ต้องเป็นหนึ่งในลิสต์นี้" เหมือนที่เราใช้ใน env.ts

ข้อสุดท้ายนี่แหละคือจุดที่ Schema ของเรายังทำได้ไม่ดี คอลัมน์ frequency ของเราเป็น varchar ที่ใส่อะไรลงไปก็ได้ ทั้งที่มันควรเลือกได้แค่ daily, weekly, monthly ถ้าเปลี่ยนเป็น Enum ข้อมูลจะสม่ำเสมอขึ้น และคนทำหน้าบ้านจะรู้ทันทีว่ามีตัวเลือกอะไรบ้าง เอาไปทำ Dropdown ได้เลย

สี่กับดักที่เจอบ่อย

ลืมใส่ Foreign Key จริง ๆ — การมีคอลัมน์ชื่อ user_id ไม่ได้แปลว่ามีความสัมพันธ์ ถ้าไม่มี .references() ฐานข้อมูลจะยอมให้ใส่ ID มั่ว ๆ ที่ไม่มีอยู่จริงลงไปได้

// ผิด — ไม่มีการตรวจสอบความถูกต้องของความสัมพันธ์
userId: uuid('user_id').notNull(),

// ถูก — ฐานข้อมูลรับประกันให้ว่า User คนนั้นมีอยู่จริง
userId: uuid('user_id')
  .references(() => users.id, { onDelete: 'cascade' })
  .notNull(),

ลืมทำ Index — Query อย่าง SELECT * FROM habits WHERE user_id = ? ถ้าไม่มี Index จะกลายเป็นการสแกนทั้งตาราง ตอนมีข้อมูลร้อยแถวไม่รู้สึกอะไร ตอนมีล้านแถวถึงจะรู้สึก

Normalize เกินจำเป็น — เคยเห็นคนแยกตาราง user_first_names ออกมาต่างหากมาแล้ว ผลคือทุก Query ต้อง JOIN เพิ่มโดยไม่ได้อะไรกลับมาเลย ชื่อจริงควรอยู่บนตาราง users ตรงนั้นแหละ

ไม่มี Timestamp — ข้อเดิมที่ต้องย้ำ เพราะพอไม่มีแล้วอยากได้ทีหลัง คุณจะเติมย้อนหลังให้ข้อมูลเก่าไม่ได้เลย

Migration แบบไหนที่ยากที่สุด

มีคนถามผมว่าจากประสบการณ์แล้ว Migration แบบไหนปวดหัวที่สุด งานแรกในชีวิตผมเคยใช้เวลาทำ Database Migration ตัวเดียวไปสองเดือน และคำตอบก็คือข้อเดิมที่เราคุยกันในบทที่ 15 คือDestructive Change

วิธีเลี่ยงคือทำให้ทุกอย่างเป็นการเพิ่ม ถ้าคุณแค่เพิ่มของใหม่โดยไม่แตะของเก่า จะไม่มีโค้ดไหนพัง เพราะยังไม่มีใครพึ่งพาของที่เพิ่งเกิด แล้วค่อยแบ่งเป็นสองรอบตามที่คุยกันไป

อีกทางที่ช่วยได้มากในยุคนี้คือ Database Branching ซึ่งบริการอย่าง Neon หรือ PlanetScale มีให้ แนวคิดคือแตก Branch ของฐานข้อมูลได้เหมือนแตก Branch ของ Git พร้อมข้อมูลข้างในทั้งก้อน ทำให้ทดลอง Migration บนสำเนาจริงได้โดยไม่แตะตัวจริง บางเจ้าตรวจให้ด้วยซ้ำว่าการเปลี่ยนแปลงนี้จะทำให้พังหรือเปล่าแล้วแนะนำทางออกให้

แน่นอนว่าการย้ายแพลตฟอร์มฐานข้อมูลของระบบที่วิ่งอยู่บน Production แล้วนั้นแทบเป็นไปไม่ได้ แต่ถ้าคุณเริ่มโปรเจกต์ใหม่วันนี้ ผมแนะนำให้เลือกเจ้าที่มี Branching ตั้งแต่แรก

พูดถึง Neon พอดี บทหน้าเราจะไปสร้าง Database จริงบนนั้นกัน แบบฟรีและไม่ต้องสมัครสมาชิกด้วยซ้ำ
