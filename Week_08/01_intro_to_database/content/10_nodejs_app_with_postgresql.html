สร้างแอป Node.js ด้วย PostgreSQL

มาเขียนโปรเจ็กต์ Node.js สั้นๆ กันอย่างรวดเร็วเพื่อช่วยคุณถ่ายทอดทักษะจากบรรทัดคำสั่งไปยังโลกแห่งการเขียนโค้ด

คุณสามารถเข้าถึงตัวอย่างทั้งหมดสำหรับโปรเจ็กต์นี้ได้ที่นี่

สร้างไดเร็กทอรีใหม่ ในไดเร็กทอรีนั้นให้รัน:

npm init -y
npm pkg set type=module
npm i pg@8.20.0 express@5.2.1
mkdir static
touch static/index.html server.js
code . # or open this folder in VS Code or whatever editor you want

เรามาสร้างส่วนหน้าโง่ๆ ที่สร้างคำค้นหากับส่วนหลังกันดีกว่า ใน static/index.html ใส่:

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>PostgreSQL Sample</title>
  </head>
  <body>
    <input type="text" placeholder="Board ID Number" id="search" />
    <button id="btn">Search</button>
    <pre>
        <code id="code"></code>
    </pre>

    <script>
      const btn = document.getElementById("btn");
      const code = document.getElementById("code");
      const search = document.getElementById("search");

      btn.addEventListener("click", async () => {
        code.innerText = "loading";

        const res = await fetch(
          "/get?search=" + encodeURIComponent(search.value),
        );
        const json = await res.json();

        code.innerText = "\n" + JSON.stringify(json, null, 4);
      });
    </script>
  </body>
</html>

ปกติแล้วฉันจะบอกว่าอย่าใส่แท็กสคริปต์ขนาดใหญ่ลงไป แต่ตอนนี้เรากำลังมุ่งสู่ความเรียบง่าย คุณสามารถปรับโครงสร้างสิ่งนี้ใหม่ในภายหลังเพื่อให้ได้สิ่งที่ดีกว่า

ใน server.js ให้ใส่สิ่งนี้:

import express from "express";
import { Pool } from "pg";
const pool = new Pool({
  connectionString: "postgresql://postgres:mysecretpassword@localhost:5432",
});

async function init() {
  const app = express();

  app.get("/get", async (req, res) => {
    const client = await pool.connect();
    const [commentsRes, boardRes] = await Promise.all([
      client.query(
        // `SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = ${req.query.search}`
        "SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = $1",
        [req.query.search],
      ),
      client.query("SELECT * FROM boards WHERE board_id = $1", [
        req.query.search,
      ]),
    ]);
    res
      .json({
        status: "ok",
        board: boardRes.rows[0] || {},
        posts: commentsRes.rows,
      })
      .end();
    client.release(); // don't use client.end()
  });

  const PORT = process.env.PORT || 3000;
  app.use(express.static("./static"));
  app.listen(PORT);

  console.log(`running on http://localhost:${PORT}`);
}
init();
  • แอปนี้มีเจตนาคล้ายกับ MongoDB ที่เราจะทำในภายหลังเพื่อแสดงให้คุณเห็นว่าการเข้าถึงฐานข้อมูลนั้นค่อนข้างคล้ายกันทั่วทั้งกระดาน
  • เรากำลังใช้พูลการเชื่อมต่อ PostgreSQL สามารถรองรับการเชื่อมต่อได้มากมายเท่านั้น และเป็นกระบวนการที่ช้าในการยกเลิกการเชื่อมต่ออย่างต่อเนื่อง สิ่งนี้จะยึดการเชื่อมต่อได้มากเท่าที่จำเป็นและนำมาใช้ซ้ำให้ได้มากที่สุด
  • เราใช้สองแบบสอบถาม นี่เป็นเรื่องปกติ ในกรณีนี้ เราสามารถเพิ่มประสิทธิภาพการค้นหาแต่ละรายการแยกกัน และอาจแคชผลลัพธ์แต่ละรายการแยกกันเพื่อประสิทธิภาพด้วย
  • ฉันขี้เกียจขอ * ที่นี่ คุณควรขอเฉพาะสิ่งที่คุณต้องการเท่านั้น
  • ด้วย rich_content มันจะสร้างแถวใหม่ในการตอบกลับสำหรับแต่ละแถว rich_content ที่ได้รับ คุณจะต้องรวมสิ่งเหล่านี้ไว้ในโค้ดซึ่งก็โอเค มีวิธีที่ง่ายกว่าในการสืบค้นเพื่อรวมไว้ล่วงหน้าหรือคุณอาจใช้การสืบค้นครั้งที่สามก็ได้

การฉีด SQL

  • เรากำลังใช้สิ่งที่เรียกว่าแบบสอบถามแบบกำหนดพารามิเตอร์กับธุรกิจ $1 เมื่อใดก็ตามที่คุณใส่เนื้อหาที่ผู้ใช้กำหนดลงในคำสั่ง SQL โดยตรง คุณจะต้องตระหนักถึงอันตรายที่ผู้ใช้สามารถใช้สิ่งนั้นในทางที่ผิดเพื่อเรียกใช้แบบสอบถามกับฐานข้อมูลของคุณ ซึ่งอาจเปิดเผยข้อมูลที่ละเอียดอ่อน (เช่น ตารางผู้ใช้ของเรา) หรือทำสิ่งที่ทำลายล้าง (เช่น ทำตารางทั้งหมดของเราตก)

สมมติว่าเราระมัดระวังน้อยลงเล็กน้อยและโค้ดของเรามีลักษณะดังนี้:

import express from "express";
import { Pool } from "pg";
const pool = new Pool({
  connectionString: "postgresql://postgres:mysecretpassword@localhost:5432",
});

async function init() {
  const app = express();

  app.get("/get", async (req, res) => {
    const client = await pool.connect();
    const [commentsRes, boardRes] = await Promise.all([
      client.query(
        `SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = ${req.query.search}`,
        // "SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = $1",
        // [req.query.search]
      ),
      client.query(
        "SELECT * FROM boards WHERE board_id = $1",
        [
          // req.query.search,
          39,
        ],
      ),
    ]);
    res
      .json({
        status: "ok",
        board: boardRes.rows[0] || {},
        // posts: commentsRes.rows,
        posts: commentsRes,
      })
      .end();
    await client.release();
  });

  app.use(express.static("./static"));
  app.listen(process.env.PORT || 3000);
}
init();
  • จะเป็นอย่างไรถ้าเราใส่ 1; SELECT * FROM users; -- เป็นคำค้นหาของเรา 1; จะตอบสนองแบบสอบถามแรก จากนั้นเราสามารถเรียกใช้แบบสอบถามที่สองเพื่อแสดงผู้ใช้ทั้งหมดในฐานข้อมูลของเรา แย่มาก แย่มาก
  • ตอนนี้ ฉันต้องเปลี่ยนโค้ดบางส่วนเพื่อให้มันแสดงข้อมูลที่กรองออกมาได้จริง แต่นั่นก็ไม่จำเป็นด้วยซ้ำ จะเป็นอย่างไรถ้าเราใส่ 1; DROP TABLE users; --? มันจะล้างฐานข้อมูลผู้ใช้ทั้งหมดของเรา! หวังว่าเราจะมีข้อมูลสำรอง
  • จริงๆ แล้วยังมีอีกมากที่พวกเขาสามารถทำได้ พวกเขาสามารถเพิ่มผู้ใช้ใหม่ใน PostgreSQL คว้าที่อยู่ IP จากนั้นเชื่อมต่อกับฐานข้อมูลเอง
  • ไม่จำเป็นต้องพูดว่า นี่เป็นหายนะถ้ามันเกิดขึ้นกับคุณ แบบสอบถามที่มีการกำหนดพารามิเตอร์ป้องกันสิ่งนี้ (วิธีที่เราเขียนไว้ด้านบน) มันจะไม่ยอมให้สิ่งที่อยู่ในคิวรีเป็นสิ่งที่ SQL สามารถแทรกแซงเป็นการดำเนินการได้
