เริ่มต้นใช้งาน PostgreSQL

บทที่แล้วเรารู้จัก Relational Database ในภาพรวมแล้ว คราวนี้มาลองใช้งาน PostgreSQL จริงกัน

เริ่ม PostgreSQL

เริ่มจากรัน PostgreSQL เวอร์ชัน 18 ใน Container ใช้เวอร์ชันเดียวกับตัวอย่างก่อนเพื่อให้คำสั่งและผลลัพธ์ตรงกัน

docker run --name my-postgres -e POSTGRES_PASSWORD=mysecretpassword -p 5432:5432 -d --rm postgres:18

docker exec -it -u postgres my-postgres psql
  • ต้องกำหนด Password ไม่เช่นนั้น PostgreSQL จะไม่เริ่มทำงาน ค่านี้ใช้เพื่อการเรียนเท่านั้น Production ต้องใช้ Password ที่ปลอดภัยกว่านี้
  • คำสั่ง docker exec เชื่อมต่อเข้าไปในฐานะผู้ใช้ postgres

ฐานข้อมูลและตาราง

คำว่า “Database” ใช้ได้สองระดับ ระดับแรกคือผลิตภัณฑ์อย่าง PostgreSQL หรือ MySQL ส่วนอีกระดับคือ Database ที่สร้างอยู่ภายใน PostgreSQL หนึ่ง Server สามารถมีหลาย Database และแต่ละ Database มีกลุ่มตารางของตัวเอง เช่น แยกข้อมูลงานวิเคราะห์ออกจากข้อมูลงานประจำวัน

ถ้าเทียบกับ Spreadsheet ให้คิดว่า Database คือไฟล์หนึ่งไฟล์ การเชื่อมข้อมูลข้ามไฟล์ทำได้แต่ไม่สะดวกนัก เราจึงมักเก็บตารางที่เกี่ยวข้องกันไว้ใน Database เดียวกัน

แอปขนาดเล็กอาจใช้ Database เดียวได้ทั้งหมด ส่วนระบบใหญ่กว่าอาจแบ่งเป็นหลาย Database ตามขอบเขตงาน เช่น แยกข้อมูลการชำระเงินออกจากข้อมูลสินค้า วิธีแบ่งขึ้นอยู่กับความต้องการและความรับผิดชอบของแต่ละส่วนในระบบ

Table คล้ายหนึ่ง Sheet ภายในไฟล์ ประกอบด้วย Row และ Column หนึ่ง Row หรือ Record แทนข้อมูลหนึ่งรายการ เช่น ผู้ใช้หนึ่งคน ส่วน Column แทน Field ของข้อมูล เช่น ชื่อหรืออีเมล

แต่ละ Table มี Schema กำหนดไว้ล่วงหน้าว่ามี Column อะไรและใช้ชนิดข้อมูลใด การเปลี่ยน Schema บนตารางใหญ่อาจใช้เวลานานและกระทบระบบ จึงควรออกแบบโครงสร้างให้รอบคอบตั้งแต่ต้น

ตอนนี้รู้ภาพรวมเท่านี้ก็พอ เดี๋ยวรายละเอียดจะชัดขึ้นเมื่อเราเริ่มสร้างตัวอย่างจริง

คำสั่ง SQL แรกของเรา

งานแรกคือสร้าง Database ใหม่ด้วย SQL แม้ SQL ของแต่ละระบบจะคล้ายกัน แต่ไม่เหมือนกันทุกจุด ดังนั้นคำสั่งของ MySQL อาจต้องปรับก่อนนำมาใช้กับ PostgreSQL

ตัวอย่างจะเขียน SQL Keyword เป็นตัวพิมพ์ใหญ่เพื่อแยกออกจากชื่อตารางและ Field ทำให้อ่าน Query ได้ง่ายขึ้น นี่เป็น Convention ที่ใช้กันทั่วไป แต่ PostgreSQL ไม่ได้บังคับ

เมื่อเปิด psql เราจะเชื่อมอยู่กับ Database เริ่มต้นชื่อ postgres ก่อน ต่อไปมาสร้าง Database ของเราเองชื่อ message_boards

รัน Query ด้านล่างและอย่าลืมใส่ ; ท้ายคำสั่ง ใน SQL เครื่องหมายนี้บอกว่าคำสั่งจบแล้ว ถ้ายังไม่ใส่ psql จะรอให้เราพิมพ์ต่อ

CREATE DATABASE message_boards;

ถ้าสำเร็จจะเห็นข้อความ CREATE DATABASE ตอนนี้เชื่อมต่อเข้า Database ใหม่นี้ได้เลย

\c message_boards;
-- \connect message_boards works too

คำสั่งที่ขึ้นต้นด้วย \ เป็นคำสั่งของ psql CLI ไม่ใช่ SQL โดย \c ใช้สลับไปยัง Database ที่ต้องการ ลองดูคำสั่ง CLI อื่นที่ใช้บ่อยกันด้วย

-- see all databases
\l

-- see all tables in this database, probably won't see anything
\d

-- see all available commands
\?

-- see available queries
\h

-- run a shell command
\! ls && echo "hi from shell!"

ใน SQL เราใช้ -- สำหรับเขียน Comment หนึ่งบรรทัด

ตารางแรก

เมื่อสร้างและเชื่อมต่อ Database แล้ว ขั้นต่อไปคือสร้าง Table แรก

CREATE TABLE users (
  user_id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  username VARCHAR ( 25 ) UNIQUE NOT NULL,
  email VARCHAR ( 50 ) UNIQUE NOT NULL,
  full_name VARCHAR ( 100 ) NOT NULL,
  last_login TIMESTAMP,
  created_on TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

มาดูรายละเอียดของคำสั่งนี้กัน

  • CREATE TABLE คือคำสั่งสำหรับสร้างตารางใหม่ โดยในที่นี้เราตั้งชื่อตารางว่า users
  • user_id เป็นฟิลด์ตัวเลขที่เพิ่มค่าอัตโนมัติ ผู้ใช้คนแรกจะมี user_id เป็น 1 คนที่สองเป็น 2 และเพิ่มต่อไปเรื่อย ๆ ซึ่งเป็นหน้าที่ของ GENERATED ALWAYS AS IDENTITY ส่วน PRIMARY KEY ระบุว่าฟิลด์นี้เป็นคีย์หลักและ PostgreSQL จะสร้างดัชนีให้โดยอัตโนมัติ
  • PostgreSQL รุ่นเก่ามักใช้ SERIAL สำหรับ ID ที่เพิ่มอัตโนมัติ แต่ GENERATED ALWAYS AS IDENTITY เป็น syntax ที่ใหม่กว่าและตรงตามมาตรฐาน SQL มากกว่า จึงควรเลือกใช้แบบใหม่นี้
  • PostgreSQL ไม่สร้าง Primary Key ให้อัตโนมัติ เราต้องประกาศเอง ต่างจาก Database บางตัวอย่าง MongoDB
  • VARCHAR ใช้เก็บ String โดยกำหนดความยาวสูงสุดไว้ username ยาวได้ไม่เกิน 25 ตัวอักษร และ email ไม่เกิน 50 ตัวอักษร ทั้งสอง Field ห้ามซ้ำด้วย UNIQUE และห้ามเป็น NULL
  • NOT NULL ยังยอมให้เก็บ String ว่างได้ เพราะ String ว่างไม่เท่ากับ NULL
  • full_name ไม่ได้กำหนด UNIQUE เพราะผู้ใช้หลายคนอาจมีชื่อซ้ำกันได้
  • last_login เก็บเวลาที่ผู้ใช้เข้าใช้ล่าสุด Field นี้เป็น NULL ได้ เพราะผู้ใช้ที่เพิ่งสมัครอาจยังไม่เคยเข้าสู่ระบบ
  • เราเพิ่มฟิลด์ created_on ไว้ด้วย หากคำสั่งไม่ได้ระบุเวลา PostgreSQL จะใช้เวลาปัจจุบันเป็นค่าเริ่มต้น แต่เรายังสามารถกำหนดเวลาอื่นเองได้ เช่น เมื่อนำเข้ารายชื่อผู้ใช้จากฐานข้อมูลอื่น
  • CURRENT_TIMESTAMP เป็น syntax มาตรฐาน SQL ส่วน NOW() เป็นฟังก์ชันของ PostgreSQL ในกรณีนี้ทั้งสองให้ผลเหมือนกัน
  • ฟิลด์ created_on ช่วยให้เราติดตามได้ว่าผู้ใช้แต่ละคนถูกสร้างขึ้นเมื่อใด

PostgreSQL ยังมี Data Type อีกมาก หากต้องการดูรายการทั้งหมดอ่านต่อได้จาก เอกสาร PostgreSQL

บันทึกแรก

มาแทรกผู้ใช้ลงในตารางใหม่ของเรา เหมือนกับการเพิ่มแถวใหม่ลงในสเปรดชีต

INSERT INTO users (username, email, full_name) VALUES ('btholt', 'lol@example.com', 'Varis Dara');
  • INSERT INTO ใช้เพิ่ม Record ใหม่ โดยตัวอย่างนี้เพิ่มข้อมูลลง Table users
  • ชื่อ Field ในวงเล็บต้องเรียงให้ตรงกับค่าที่ส่งไว้ใน VALUES
  • เราไม่ต้องส่ง user_id เพราะ PostgreSQL สร้างค่าให้เอง
  • เราไม่ส่ง last_login เพราะ Field นี้ไม่บังคับ และผู้ใช้ใหม่ยังไม่เคยเข้าสู่ระบบ

ถ้าสำเร็จจะเห็น INSERT 0 1 เลข 1 หมายถึงเพิ่มข้อมูลหนึ่ง Record ส่วนเลข 0 คือ OID ซึ่งคอร์สนี้ไม่ได้ใช้ จึงไม่ต้องกังวลกับค่านี้

ดูบันทึกแรกของคุณ

ก่อนเรียน Query แบบอื่น ลองดึงทุก Record และทุก Field ออกจาก Table เพื่อดูข้อมูลที่เพิ่งเพิ่มกันก่อน

SELECT * FROM users;

ผลลัพธ์ควรมีหนึ่ง Row ที่เราเพิ่งเพิ่มไว้ บทถัดไปจะอธิบาย Query นี้และต่อยอดไปยังการค้นหารูปแบบอื่น
