การใช้ JSON ใน PostgreSQL

เจสันบี

บางครั้งคุณมีข้อมูลที่ไม่มีสคีมาที่ดี หากคุณพยายามที่จะใส่ลงในฐานข้อมูลตาราง เช่น PostgreSQL คุณจะพบว่ามีชื่อฟิลด์ที่กว้างมากซึ่งจะต้องตีความด้วยโค้ด ไม่เช่นนั้น คุณจะจบลงด้วยตารางหลายตารางเพื่ออธิบายสคีมาที่แตกต่างกัน นี่คือที่เดียวที่ฐานข้อมูลตามเอกสารอย่าง MongoDB โดดเด่นมาก ฐานข้อมูลแบบไม่มีสคีมาทำงานได้ดีในสถานการณ์เหล่านี้

อย่างไรก็ตาม PostgreSQL มีพลังวิเศษที่นี่: ประเภทข้อมูล JSONB ซึ่งจะทำให้คุณสามารถใส่ออบเจ็กต์ JSONB ลงในคอลัมน์ จากนั้นคุณสามารถใช้ SQL เพื่อสืบค้นออบเจ็กต์เหล่านั้นได้

เรามาเป็นตัวอย่างสำหรับกระดานข้อความของเรา คุณต้องการเพิ่มคุณสมบัติใหม่ที่อนุญาตให้ผู้ใช้ทำการฝังเนื้อหาที่หลากหลายในกระดานข้อความของคุณ สำหรับผู้เริ่มต้น พวกเขาสามารถฝังโพล รูปภาพ และวิดีโอได้ แต่คุณสามารถจินตนาการได้ว่ามันจะเพิ่มมากขึ้นในอนาคต เพื่อให้สามารถฝังทวีต เอกสาร และสิ่งอื่นๆ ที่เรายังไม่ได้ฝันถึงได้ คุณต้องการรักษาความยืดหยุ่นในอนาคตไว้

สิ่งนี้อาจเป็นไปได้ที่จะสร้างแบบจำลองด้วยสคีมาปกติ แต่ออกมาค่อนข้างน่าเกลียดและเข้าใจยาก และเป็นไปไม่ได้ที่จะคาดการณ์แผนการเติบโตในอนาคตทั้งหมดของเราในตอนนี้ นี่คือจุดที่ชนิดข้อมูล JSONB กำลังจะโดดเด่นจริงๆ นี่คือคำค้นหาที่เราดำเนินการเพื่อสร้าง (คุณไม่จำเป็นต้องเรียกใช้อีกครั้ง)

DROP TABLE IF EXISTS rich_content;

CREATE TABLE rich_content (
  content_id INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  comment_id INT REFERENCES comments(comment_id) ON DELETE CASCADE,
  content JSONB NOT NULL
);

INSERT INTO rich_content
  (comment_id, content)
VALUES
  (63, '{ "type": "poll", "question": "What is your favorite color?", "options": ["blue", "red", "green", "yellow"] }'),
  (71, '{ "type": "video", "url": "https://youtu.be/dQw4w9WgXcQ", "dimensions": { "height": 1080, "width": 1920 }}'),
  (104, '{ "type": "poll", "question": "Is this your favorite video?", "options": ["yes", "no", "oh you"] }'),
  (201, '{ "type": "image", "url": "https://btholt.github.io/complete-intro-to-linux-and-the-cli/WORDMARK-Small.png", "dimensions": { "height": 400, "width": 1084 }}'),
  (274, '{ "type": "image", "url": "https://btholt.github.io/complete-intro-to-linux-and-the-cli/HEADER.png", "dimensions": { "height": 237 , "width": 3301 }}');
  • ประเภทข้อมูล JSONB คือดาวที่ส่องแสงที่นี่ ซึ่งช่วยให้เราสามารถแทรกออบเจ็กต์ JSON เพื่อสอบถามในภายหลังได้
  • PostgreSQL จะไม่ยอมให้คุณแทรก JSON ที่มีรูปแบบไม่ถูกต้อง ดังนั้นเราจะตรวจสอบความถูกต้องให้กับคุณ
  • สังเกตว่าคุณสามารถมีรังได้มากเท่าที่คุณต้องการ JSON ที่ถูกต้องใดๆ จะใช้ได้ที่นี่

เลยมาสอบถามกันหน่อย! ตั้งแต่ PostgreSQL 14 คุณสามารถใช้เครื่องหมายวงเล็บได้เหมือนกับที่คุณใช้ใน JavaScript:

SELECT content['type'] FROM rich_content;

คุณจะได้รับสิ่งนี้:

## ?column?

"poll"
"video"
"poll"
"image"
"image"

สังเกตเครื่องหมายคำพูดรอบๆ ค่า—PostgreSQL ส่งคืนค่าเหล่านี้เป็น JSON ไม่ใช่ข้อความธรรมดา ความแตกต่างนั้นสำคัญเมื่อคุณต้องการเปรียบเทียบค่าหรือกรองผลลัพธ์ที่ซ้ำกัน

มันทำโพลและรูปภาพซ้ำสองครั้งเพราะมีสองอัน จะเป็นอย่างไรถ้าเราเพียงต้องการตัวเลือกที่แตกต่างและไม่ทำซ้ำ? GROUP BY คงจะได้ผล แต่ขออ้อมไปพูดถึง SELECT DISTINCT ดีกว่า SELECT DISTINCT จะกรองผลลัพธ์ที่ซ้ำกันให้คุณ ลองสิ่งนี้ (จะเกิดข้อผิดพลาด):

SELECT DISTINCT content['type'] FROM rich_content;

PostgreSQL ไม่ทราบว่าข้อมูลประเภทใดที่จะได้รับกลับมาจาก JSON ดังนั้นจึงปฏิเสธที่จะทำการเปรียบเทียบกับผลลัพธ์ใดๆ เราจำเป็นต้องแยกค่าเป็นข้อความ และเพื่อที่เราจะใช้ตัวดำเนินการ ->>:

SELECT DISTINCT content ->> 'type' FROM rich_content;

->> ระบุว่า "ให้ค่านี้เป็นสตริงข้อความธรรมดาแก่ฉัน" ตอนนี้ PostgreSQL ก็สามารถเปรียบเทียบได้แล้ว

จะเป็นอย่างไรหากเราต้องการค้นหาเฉพาะโพลเท่านั้น

SELECT content ->> 'type' AS content_type, comment_id 
FROM rich_content 
WHERE content ->> 'type' = 'poll';

น่าเสียดายเนื่องจากลำดับการดำเนินการ (ซึ่งเกิดขึ้นก่อน SELECT) คุณไม่สามารถอ้างอิง content_type และต้องให้นิพจน์แบบเต็ม

โอเค อันสุดท้าย จะเป็นอย่างไรหากเราต้องการหาความกว้างและความสูงทั้งหมด? ที่นี่เราสามารถผสมเครื่องหมายวงเล็บกับ ->> ได้:

SELECT
  content['dimensions'] ->> 'height' AS height,
  content['dimensions'] ->> 'width' AS width,
  comment_id
FROM
  rich_content;

ใช้วงเล็บเพื่อนำทางไปยังวัตถุที่ซ้อนกัน จากนั้น ->> ที่ส่วนท้ายเมื่อคุณต้องการค่าเป็นข้อความ สิ่งนี้จะคืนอันที่ไม่มีความสูงและความกว้างกลับมาให้คุณ หากต้องการกรองสิ่งเหล่านั้นออก:

SELECT
  content['dimensions'] ->> 'height' AS height,
  content['dimensions'] ->> 'width' AS width,
  comment_id
FROM
  rich_content
WHERE
  content['dimensions'] IS NOT NULL;

ตัวดำเนินการ JSON และการแยกข้อความ

คุณจะพบวิธีการต่างๆ ในการรับข้อความจากคอลัมน์ JSON นี่คือสิ่งที่คุณจะเห็นในโค้ดเบส:

->> (พบบ่อยที่สุด)

แยกค่าเป็นข้อความ นี่คือสิ่งที่รหัสการผลิตส่วนใหญ่ใช้:

SELECT content ->> 'type' FROM rich_content;

-> (ส่งคืน JSON)

แยกค่าแต่คงไว้เป็นประเภท JSON มีประโยชน์สำหรับการเชื่อมโยงหรือเมื่อคุณต้องการรักษาโครงสร้าง JSON:

SELECT content -> 'dimensions' -> 'height' FROM rich_content;

สัญลักษณ์วงเล็บ (PostgreSQL 14+)

ไวยากรณ์สมัยใหม่ที่ให้ความรู้สึกเหมือน JavaScript ส่งคืน JSON ไม่ใช่ข้อความ:

SELECT content['dimensions']['height'] FROM rich_content;

หล่อ

แปลง JSON เป็นข้อความอย่างชัดเจน ละเอียดมากขึ้นแต่ชัดเจนมากเกี่ยวกับเจตนา:

SELECT CAST(content['type'] AS TEXT) FROM rich_content;

#>> พร้อมเส้นทางว่าง

เคล็ดลับที่คุณอาจเห็นเป็นครั้งคราว—แยกเป็นข้อความจากค่า JSON:

SELECT content['type'] #>> '{}' FROM rich_content;

ในทางปฏิบัติ ทีมส่วนใหญ่ใช้วงเล็บเพื่อนำทาง JSON และ ->> ที่ซ้อนกัน เมื่อต้องการแยกข้อความ รูปแบบลูกโซ่ -> และ ->> รุ่นเก่ายังคงใช้งานได้ดี และคุณจะเห็นได้มากมายในโค้ดที่มีอยู่:

-- Older style (still common)
SELECT content -> 'dimensions' ->> 'height' FROM rich_content;

-- Modern mixed style
SELECT content['dimensions'] ->> 'height' FROM rich_content;

ทั้งสองถูกต้อง ใช้อะไรก็ได้ที่ทีมของคุณต้องการ
