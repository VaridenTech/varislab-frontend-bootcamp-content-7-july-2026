สร้างแอป Node.js ด้วย Neo4j

มาเขียนโปรเจ็กต์ Node.js สั้นๆ กันอย่างรวดเร็วเพื่อช่วยคุณถ่ายทอดทักษะจากบรรทัดคำสั่งไปยังโลกแห่งการเขียนโค้ด

คุณสามารถเข้าถึงตัวอย่างทั้งหมดสำหรับโปรเจ็กต์นี้ได้ที่นี่

สร้างไดเร็กทอรีใหม่ ในไดเร็กทอรีนั้นให้รัน:

npm init -y
npm pkg set type=module
npm i neo4j-driver@6.0.1 express@5.2.1
mkdir static
touch static/index.html server.js
code . # or open this folder in VS Code or whatever editor you want

เรามาสร้างส่วนหน้าโง่ๆ ที่สร้างคำค้นหากับส่วนหลังกันดีกว่า ใน static/index.html ใส่:

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Neo4j Sample</title>
  </head>
  <body>
    <input type="text" placeholder="Person 1" id="person1" />
    <input type="text" placeholder="Person 2" id="person2" />
    <button id="btn">Search</button>
    <pre>
        <code id="code"></code>
    </pre>

    <script>
      const btn = document.getElementById("btn");
      const code = document.getElementById("code");
      const person1 = document.getElementById("person1");
      const person2 = document.getElementById("person2");

      btn.addEventListener("click", async () => {
        code.innerText = "loading";

        const res = await fetch(
          `/get?person1=${encodeURIComponent(
            person1.value,
          )}&person2=${encodeURIComponent(person2.value)}`,
        );
        const json = await res.json();

        code.innerText = "\n" + JSON.stringify(json, null, 4);
      });
    </script>
  </body>
</html>
  • ปกติแล้วฉันจะบอกว่าอย่าใส่แท็กสคริปต์ขนาดใหญ่ลงไป แต่ตอนนี้เรากำลังมุ่งสู่ความเรียบง่าย คุณสามารถปรับโครงสร้างสิ่งนี้ใหม่ในภายหลังเพื่อให้ได้สิ่งที่ดีกว่า
  • จะดีกว่ามากถ้าเรามีนักแสดง นักแสดง และผู้กำกับทั้งหมดในกราฟแบบเลื่อนลง แทนที่จะเป็นข้อความรูปแบบอิสระ ฉันจะปล่อยให้คุณทำแบบสอบถามและอัปเดต UI

ใน server.js ให้ใส่สิ่งนี้:

import express from "express";
import neo4j from "neo4j-driver";

const CONNECTION_STRING =
  process.env.NEO4J_CONNECTION_STRING || "bolt://localhost:7687";

const driver = neo4j.driver(CONNECTION_STRING);

async function init() {
  const app = express();

  app.get("/get", async (req, res) => {
    const session = driver.session();
    const result = await session.run(
      `
        MATCH path = shortestPath(
            (First:Person {name: $person1 })-[*]-(Second:Person {name: $person2 })
        )
        UNWIND nodes(path) as node
        RETURN coalesce(node.name, node.title) as text;
    `,
      {
        person1: req.query.person1,
        person2: req.query.person2,
      },
    );

    res.json({
      status: "ok",
      path: result.records.map((record) => record.get("text")),
    });

    await session.close();
  });

  app.use(express.static("./static"));
  const PORT = process.env.PORT || 3000;
  app.listen(PORT);
  console.log(`running on http://localhost:${PORT}`);
}
init();
  • Neo4j มีโปรโตคอลการถ่ายโอนของตัวเอง (ตรงข้ามกับ HTTP) ที่เรียกว่า bolt ทั้ง PostgreSQL และ MongoDB ก็มีของตัวเองเช่นกัน
  • เช่นเดียวกับพูลของ PostgreSQL ไดรเวอร์ของ Neo4j จะทำสิ่งนั้นเพื่อคุณและให้ไคลเอนต์แก่คุณผ่านเซสชัน ตรวจสอบให้แน่ใจว่าคุณปิดเซสชันเมื่อดำเนินการเสร็จแล้วเพื่อให้สามารถใช้การเชื่อมต่อซ้ำได้
  • เรากำลังทำการสืบค้นแบบกำหนดพารามิเตอร์เช่นเดียวกับ PostgreSQL อย่างที่คุณเห็นว่าคุณกำลังส่งสตริงไปที่ Neo4j ซึ่งเป็นการตีความสตริงนั้นเป็นคำสั่ง ทุกครั้งที่เกิดเหตุการณ์เช่นนี้ คุณอาจถูกฉีดยาโจมตีได้ โชคดีที่คุณสามารถทำสิ่งเดียวกันได้ที่นี่ด้วยการสืบค้นแบบกำหนดพารามิเตอร์ ดังนั้นไดรเวอร์จึงมั่นใจได้ว่าจะไม่มีสิ่งใดเสียหายเกิดขึ้น
  • โดยพื้นฐานแล้วนี่เป็นคำค้นหาของ Kevin Bacon แต่โดยทั่วไปแล้วจะเป็นคำค้นหาแบบคนสองคน
  • Neo4j ส่งคืนรายการของวัตถุ Record และคุณต้องเรียก get() ในแต่ละระเบียนเพื่อรับแอตทริบิวต์ที่คุณต้องการ ซึ่งเป็นสิ่งที่เราทำกับคำสั่ง map เพื่อรับชื่อและหัวเรื่องแต่ละรายการ
