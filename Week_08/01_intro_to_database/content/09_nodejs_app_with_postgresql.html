สร้างแอป Node.js ด้วย PostgreSQL

เราใช้ PostgreSQL ผ่าน Command Line มาแล้ว ต่อไปมาลองเชื่อมต่อจากแอป Node.js ขนาดเล็กเพื่อเห็นภาพการใช้งานในโค้ดจริง

ดู Source Code ตัวอย่างทั้งหมด ได้ที่นี่

สร้าง Directory ใหม่ แล้วรันคำสั่งต่อไปนี้ภายใน Directory นั้น

npm init -y
npm pkg set type=module
npm i pg@8.20.0 express@5.2.1
mkdir static
touch static/index.html server.js
code . # or open this folder in VS Code or whatever editor you want

เริ่มจาก Frontend แบบเรียบง่ายที่ส่ง Request ไปยัง Backend สร้างไฟล์ static/index.html แล้วใส่โค้ดต่อไปนี้

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>PostgreSQL Sample</title>
  </head>
  <body>
    <input type="text" placeholder="Board ID Number" id="search" />
    <button id="btn">Search</button>
    <pre>
        <code id="code"></code>
    </pre>

    <script>
      const btn = document.getElementById("btn");
      const code = document.getElementById("code");
      const search = document.getElementById("search");

      btn.addEventListener("click", async () => {
        code.innerText = "loading";

        const res = await fetch(
          "/get?search=" + encodeURIComponent(search.value),
        );
        const json = await res.json();

        code.innerText = "\n" + JSON.stringify(json, null, 4);
      });
    </script>
  </body>
</html>

ในโปรเจกต์จริงควรแยก JavaScript ออกจาก HTML ให้เป็นระเบียบ แต่ตัวอย่างนี้รวมไว้ในไฟล์เดียวเพื่อให้เห็น Flow ได้ง่าย คุณค่อย Refactor ภายหลังได้

จากนั้นสร้าง server.js แล้วใส่โค้ดนี้

import express from "express";
import { Pool } from "pg";
const pool = new Pool({
  connectionString: "postgresql://postgres:mysecretpassword@localhost:5432",
});

async function init() {
  const app = express();

  app.get("/get", async (req, res) => {
    const client = await pool.connect();
    const [commentsRes, boardRes] = await Promise.all([
      client.query(
        // `SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = ${req.query.search}`
        "SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = $1",
        [req.query.search],
      ),
      client.query("SELECT * FROM boards WHERE board_id = $1", [
        req.query.search,
      ]),
    ]);
    res
      .json({
        status: "ok",
        board: boardRes.rows[0] || {},
        posts: commentsRes.rows,
      })
      .end();
    client.release(); // don't use client.end()
  });

  const PORT = process.env.PORT || 3000;
  app.use(express.static("./static"));
  app.listen(PORT);

  console.log(`running on http://localhost:${PORT}`);
}
init();
  • โครงสร้างแอปตั้งใจให้คล้ายตัวอย่าง MongoDB ในบทถัดไป เพื่อให้เห็นว่า Flow การเรียก Database จาก Node.js ไม่ได้ต่างกันมาก
  • เราใช้ Connection Pool เพื่อเปิด Connection เท่าที่จำเป็นและนำกลับมาใช้ซ้ำ การเปิดและปิด Connection ใหม่ทุก Request ช้าและยังทำให้ PostgreSQL ใช้ Connection จนเต็มได้ง่าย
  • ตัวอย่างใช้สอง Query ซึ่งเป็นเรื่องปกติ และยังเปิดให้ Optimize หรือ Cache ผลลัพธ์ของแต่ละ Query แยกกันได้
  • ตัวอย่างใช้ * เพื่อให้โค้ดสั้น แต่ในงานจริงควรเลือกเฉพาะ Field ที่ต้องใช้
  • ถ้า Comment มี rich_content หลายรายการ ผลลัพธ์จะมีหลาย Row เราต้องรวม Row เหล่านั้นในโค้ด หรือปรับ Query ให้ Aggregate ข้อมูลไว้ก่อน

SQL Injection

  • $1 คือ Placeholder ของ Parameterized Query ห้ามนำค่าจากผู้ใช้มาต่อเป็น SQL โดยตรง เพราะผู้โจมตีอาจแทรกคำสั่งเพื่ออ่านข้อมูลสำคัญหรือลบ Table ได้

ลองดูตัวอย่างที่ไม่ปลอดภัย ซึ่งนำค่าจาก URL มาต่อกับ SQL โดยตรง

import express from "express";
import { Pool } from "pg";
const pool = new Pool({
  connectionString: "postgresql://postgres:mysecretpassword@localhost:5432",
});

async function init() {
  const app = express();

  app.get("/get", async (req, res) => {
    const client = await pool.connect();
    const [commentsRes, boardRes] = await Promise.all([
      client.query(
        `SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = ${req.query.search}`,
        // "SELECT * FROM comments NATURAL LEFT JOIN rich_content WHERE board_id = $1",
        // [req.query.search]
      ),
      client.query(
        "SELECT * FROM boards WHERE board_id = $1",
        [
          // req.query.search,
          39,
        ],
      ),
    ]);
    res
      .json({
        status: "ok",
        board: boardRes.rows[0] || {},
        // posts: commentsRes.rows,
        posts: commentsRes,
      })
      .end();
    await client.release();
  });

  app.use(express.static("./static"));
  app.listen(process.env.PORT || 3000);
}
init();
  • ถ้าส่ง 1; SELECT * FROM users; -- คำสั่งแรกจะทำงานตามปกติ แล้วตามด้วย Query ที่พยายามอ่านผู้ใช้ทั้งหมด
  • ค่าที่อันตรายกว่าอย่าง 1; DROP TABLE users; -- อาจลบ Table ผู้ใช้ทั้งชุดได้ หาก Account ที่แอปใช้มีสิทธิ์มากพอ
  • ผู้โจมตียังอาจสร้าง Account ใน PostgreSQL หรือเปิดทางให้เชื่อม Database จากภายนอกได้ ผลกระทบจึงไม่ได้จบแค่ข้อมูลรั่ว
  • Parameterized Query ป้องกันปัญหานี้ด้วยการส่ง Input เป็นข้อมูลแยกจาก SQL Database จึงไม่ตีความค่าของผู้ใช้เป็นคำสั่ง
