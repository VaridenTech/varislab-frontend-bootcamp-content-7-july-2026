สร้างแอป Node.js ด้วย MongoDB

มาเขียนโปรเจ็กต์ Node.js สั้นๆ กันอย่างรวดเร็วเพื่อช่วยคุณถ่ายทอดทักษะจากบรรทัดคำสั่งไปยังโลกแห่งการเขียนโค้ด

คุณสามารถเข้าถึงตัวอย่างทั้งหมดสำหรับโปรเจ็กต์นี้ได้ที่นี่

สร้างไดเร็กทอรีใหม่ ในไดเร็กทอรีนั้นให้รัน:

npm init -y
npm pkg set type=module
npm i mongodb@7.1.0 express@5.2.1
mkdir static
touch static/index.html server.js
code . # or open this folder in VS Code or whatever editor you want

เรามาสร้างส่วนหน้าโง่ๆ ที่สร้างคำค้นหากับส่วนหลังกันดีกว่า ใน static/index.html ใส่:

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>MongoDB Sample</title>
  </head>
  <body>
    <input type="text" placeholder="Search" id="search" />
    <button id="btn">Search</button>
    <pre>
        <code id="code"></code>
    </pre>

    <script>
      const btn = document.getElementById("btn");
      const code = document.getElementById("code");
      const search = document.getElementById("search");

      btn.addEventListener("click", async () => {
        code.innerText = "loading";

        const res = await fetch(
          "/get?search=" + encodeURIComponent(search.value),
        );
        const json = await res.json();

        code.innerText = "\n" + JSON.stringify(json, null, 4);
      });
    </script>
  </body>
</html>

ปกติแล้วฉันจะบอกว่าอย่าใส่แท็กสคริปต์ขนาดใหญ่ลงไป แต่ตอนนี้เรากำลังมุ่งสู่ความเรียบง่าย คุณสามารถปรับโครงสร้างสิ่งนี้ใหม่ในภายหลังเพื่อให้ได้สิ่งที่ดีกว่า

ใน server.js ให้ใส่สิ่งนี้:

import express from "express";
import { MongoClient } from "mongodb";

const connectionString =
  process.env.MONGODB_CONNECTION_STRING || "mongodb://localhost:27017";

async function init() {
  const client = new MongoClient(connectionString);
  await client.connect();

  const app = express();

  app.get("/get", async (req, res) => {
    const db = await client.db("adoption");
    const collection = db.collection("pets");

    const pets = await collection
      .find(
        {
          $text: { $search: req.query.search },
        },
        { projection: { _id: 0 } },
      )
      .sort({ score: { $meta: "textScore" } })
      .limit(10)
      .toArray();

    res.json({ status: "ok", pets }).end();
  });

  const PORT = process.env.PORT || 3000;
  app.use(express.static("./static"));
  app.listen(PORT);

  console.log(`running on http://localhost:${PORT}`);
}
init();

🚨 ปัญหาที่อาจเกิดขึ้นที่คุณอาจเห็น: หากคุณเห็นข้อผิดพลาด UnhandledPromiseRejectionWarning: MongoError: text index required for $text query (no such collection 'adoption.pets') แสดงว่าคุณอาจมีชื่อฐานข้อมูลที่ไม่ถูกต้องในรหัสของคุณ ในตัวอย่างของฉัน ฉันเรียกฐานข้อมูลของฉัน adoption หากคุณไม่ได้เปลี่ยนชื่อฐานข้อมูลของคุณ ฐานข้อมูลจะเรียกว่า test หากคุณไม่ทราบชื่อฐานข้อมูลของคุณ ให้รัน db ใน mongo shell ของคุณ แล้วระบบจะแจ้งให้คุณทราบ เมื่อคุณค้นพบชื่อฐานข้อมูลของคุณแล้ว ให้เปลี่ยนบรรทัด const db = await client.db("<the name of your database here>"); เพื่อให้ตรงกับชื่อฐานข้อมูลของคุณ

มาดูบันทึกย่อบางส่วนที่นี่ เพื่อให้ชัดเจน นี่เป็นการแสดงให้คุณเห็นถึงวิธีเชื่อมต่อกับ MongoDB จากโค้ดเป็นส่วนใหญ่ และเพื่อแสดงให้คุณเห็นว่าข้อความค้นหาทั้งหมดที่คุณได้เรียนรู้ในส่วนที่แล้วนำไปใช้ได้เกือบจะโดยไม่มีการแก้ไขใดๆ (มีความแตกต่างเล็กน้อยอยู่บ้าง)

  • เราดึงสตริงการเชื่อมต่อ MongoDB ออกจากตัวแปรสภาพแวดล้อม (ดังนั้นคุณสามารถทำได้หากทำงานบนเซิร์ฟเวอร์บนคลาวด์) ไม่เช่นนั้นจะใช้สำเนาที่รันในเครื่องเหมือนที่เราทำตอนนี้
  • คุณต้องเชื่อมต่อกับ DB จากนั้นเชื่อมต่อกับ db (ฐานข้อมูล) จากนั้นจึงเชื่อมต่อกับคอลเลกชัน
  • คุณเรียกใช้แบบสอบถามทั้งหมดในระดับคอลเลกชัน
  • ค้นหา, จัดเรียง, จำกัด, toArray, DeleteOne, findOneAndReplace ฯลฯ ทั้งหมดนี้ทำงานในแบบที่คุณคาดหวัง

แค่นั้นแหละ! นั่นเป็นเซิร์ฟเวอร์ Node.js ที่รวดเร็วที่เชื่อมต่อกับ MongoDB!
